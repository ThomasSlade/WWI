# Serbian Vojvodina declares independence
country_event = {
	id = wwi_austria.9
	title = wwi_austria.9.t
	desc = {
		text = wwi_austria.9.d.a
		trigger = {
			check_variable = { SER.serb_vojvodina_armament < 2 }
		}
	}
	desc = {
		text = wwi_austria.9.d.b
		trigger = {
			check_variable = { SER.serb_vojvodina_armament = 2 }
		}
	}
	desc = {
		text = wwi_austria.9.d.c
		trigger = {
			check_variable = { SER.serb_vojvodina_armament > 2 }
		}
	}
	picture = GFX_report_event_german_speech

	is_triggered_only = yes
	trigger = {
		AND = {
			has_idea = AUS_vojvodina_separatists
			any_owned_state = { is_core_of = SVO }
		}
	}
	fire_only_once = yes

	mean_time_to_happen = {
		days = 1
	}

	immediate = {
		hidden_effect = {
			release = SVO
			set_global_flag = serb_vojvodina_revolted
			
			SER = {
				diplomatic_relation = {
					country = SVO
					relation = puppet
					active = yes
				}
			}

			SVO = {
				inherit_technology = SER

				if = {
					limit = { check_variable = { SER.serb_vojvodina_armament < 1 } }
					load_oob = "SVO_Strength_0"
				}
				else_if = {
					limit = { check_variable = { SER.serb_vojvodina_armament < 2 } }
					load_oob = "SVO_Strength_1"
				}
				else_if = {
					limit = { check_variable = { SER.serb_vojvodina_armament < 3 } }
					load_oob = "SVO_Strength_2"
				}
				else = {
					load_oob = "SVO_Strength_3"
				}
			}

			remove_ideas = AUS_vojvodina_separatists
		}
	}

	# Declare war on Vojvodina
	option = {
		name = wwi_austria.9.a
		ai_chance = {
			base = 1
		}

		declare_war_on = {
			target = SVO
			type = take_core_state
		}
	}

	# Let them go
	option = {
		trigger = {
			NOT = { has_war_with = SER }
		}

		name = wwi_austria.9.b
	}
}

			base = 1
		}

		create_wargoal = {

			target = MNT
		}
		SER = {
			transfer_state = 224
			remove_state_core = 146
			remove_state_core = 169
		}
		MNT  = {
			add_opinion_modifier = {
				target = SER
				modifier = MNT_betrayed_by_serbia
			}
		}

		# Unless Montenegro has major guarantees, AI Austria declares war on them.
		if = {
			limit = {
				AND = {
					is_ai = yes
					NOT = {
						any_other_country = {
							is_major = yes
							has_guaranteed = MNT
						}
					}
				}
			}

			declare_war_on = {
				target = MNT
				type = annex_everything
			}
		}
	}

	# Decline
	option = {
		name = wwi_austria.10.b
		ai_chance = {
			base = 0
		}
	}
}

# Hidden event to select the province losing Obrenovic sympathies. To be fired in the scope of that province's controller (Serbia).
country_event = {
	id = events_AUS_foment_obrenovic_sympathies.4
	title = events_AUS_foment_obrenovic_sympathies.4.t
	desc = events_AUS_foment_obrenovic_sympathies.4.d
	picture = GFX_report_event_german_speech

	is_triggered_only = yes
	hidden = yes

	immediate = {
		log = "Finding state most recently flagged as fomented Obrenovic sympathies."

		set_temp_variable = { last_flagged_order = -1 }
		for_each_scope_loop = {
			array = owned_controlled_states

			if = {
				limit = {
					has_variable = fomented_sympathies_order
					check_variable = { fomented_sympathies_order > last_flagged_order }
				}
				set_temp_variable = { last_flagged_order = fomented_sympathies_order }
				set_temp_variable = { last_flagged = THIS }
			}
		}

		if = {
			limit = { NOT = { check_variable = { last_flagged = 0 } } }
			log = "Most recent Obrenovic sympathiser state found was: [?last_flagged.GetName]."
			var:last_flagged = {
				effect_AUS_remove_obrenovic_sympathies_in_state = yes

				save_event_target_as = obrenovic_sympathiser_province
				owner = { country_event = { id = events_AUS_foment_obrenovic_sympathies.3 } }
				save_event_target_as = obrenovic_sympathiser_province
				AUS = {
					country_event = { id = events_AUS_foment_obrenovic_sympathies.3 }
				}
			}
		}
		else = {
			log = "No states owned by [SER.GetAdjective] found with Obrenovic sympathies.."
		}
	}
}
