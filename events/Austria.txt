# Assassination of Franz Ferdinand
country_event = {
	id = wwi_austria.1
	title = wwi_austria.1.t
	desc = wwi_austria.1.d

	fire_only_once = yes

	is_triggered_only = yes

	immediate = {
		hidden_effect = {
			set_country_flag = AUS_ferdinand_dead
			remove_ideas = franz_ferdinand
		}
	}

	# Ultimatum.
	option = {
		name = wwi_austria.1.a
		ai_chance = { factor = 100 }
		SER = { country_event = { id = wwi_serbia.1 } }
	}

	# Back down.
	option = {
		name = wwi_austria.1.b
		ai_chance = { factor = 0 }
		add_stability = -0.1
		add_timed_idea = {
			idea = AUS_serbian_punching_bag
			days = 730
		}
	}
}

# Serbian Vojvodina declares independence
country_event = {
	id = wwi_austria.9
	title = wwi_austria.9.t
	desc = {
		text = wwi_austria.9.d.a
		trigger = {
			check_variable = { SER.serb_vojvodina_armament < 2 }
		}
	}
	desc = {
		text = wwi_austria.9.d.b
		trigger = {
			check_variable = { SER.serb_vojvodina_armament = 2 }
		}
	}
	desc = {
		text = wwi_austria.9.d.c
		trigger = {
			check_variable = { SER.serb_vojvodina_armament > 2 }
		}
	}
	picture = GFX_report_event_german_speech

	is_triggered_only = yes
	trigger = {
		AND = {
			has_idea = AUS_vojvodina_separatists
			any_owned_state = { is_core_of = SVO }
		}
	}
	fire_only_once = yes

	mean_time_to_happen = {
		days = 1
	}

	immediate = {
		hidden_effect = {
			release = SVO
			set_global_flag = serb_vojvodina_revolted
			
			SER = {
				diplomatic_relation = {
					country = SVO
					relation = puppet
					active = yes
				}
			}

			SVO = {
				inherit_technology = SER

				if = {
					limit = { check_variable = { SER.serb_vojvodina_armament < 1 } }
					load_oob = "SVO_Strength_0"
				}
				else_if = {
					limit = { check_variable = { SER.serb_vojvodina_armament < 2 } }
					load_oob = "SVO_Strength_1"
				}
				else_if = {
					limit = { check_variable = { SER.serb_vojvodina_armament < 3 } }
					load_oob = "SVO_Strength_2"
				}
				else = {
					load_oob = "SVO_Strength_3"
				}
			}

			remove_ideas = AUS_vojvodina_separatists
		}
	}

	# Declare war on Vojvodina
	option = {
		name = wwi_austria.9.a
		ai_chance = {
			base = 1
		}

		declare_war_on = {
			target = SVO
			type = take_core_state
		}
	}

	# Let them go
	option = {
		trigger = {
			NOT = { has_war_with = SER }
		}

		name = wwi_austria.9.b
	}
}

# Serbia asks for a slice of Bosnia
country_event = {
	id = wwi_austria.10
	title = wwi_austria.10.t
	desc = wwi_austria.10.d
	picture = GFX_report_event_german_speech

	is_triggered_only = yes

	# Accept
	option = {
		name = wwi_austria.10.a
		ai_chance = {
			base = 1
		}

		remove_state_core = 224
		create_wargoal = {
			type = annex_everything
			target = MNT
		}
		SER = {
			transfer_state = 224
			remove_state_core = 146
			remove_state_core = 169
		}
		MNT  = {
			add_opinion_modifier = {
				target = SER
				modifier = MNT_betrayed_by_serbia
			}
		}

		# Unless Montenegro has major guarantees, AI Austria declares war on them.
		if = {
			limit = {
				AND = {
					is_ai = yes
					NOT = {
						any_other_country = {
							is_major = yes
							has_guaranteed = MNT
						}
					}
				}
			}

			declare_war_on = {
				target = MNT
				type = annex_everything
			}
		}
	}

	# Decline
	option = {
		name = wwi_austria.10.b
		ai_chance = {
			base = 0
		}
	}
}
