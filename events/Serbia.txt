			transfer_state = 224

# After the Balkan War, whichever country is the selected arbitrator gets the event to give macedonian border-states between Bulg and Serb to whoever it favours.
# The AI will always award states to who they have the highest opinion of.
country_event = {
	id = wwi_serbia.13
	title = wwi_serbia.13.t
	desc = wwi_serbia.13.d

	is_triggered_only = yes

	immediate = {
		if = {
			limit = {
				check_variable = {
					var = ROOT.opinion@SER
					value = ROOT.opinion@BUL
					compare = greater_than_or_equals
				}
			}
			set_variable = { var = macedonian_arbitrators_favourite value = SER }
		}
		else = {
			set_variable = { var = macedonian_arbitrators_favourite value = BUL }
		}

		log = "Determined Macednian Arbitrator [?ROOT.GetName]'s preferred Balkan client: [?var:macedonian_arbitrators_favourite.GetName]"
	}

	# Rule in Serbia's favour.
	option = {
		name = wwi_serbia.13.a

		ai_chance = {
			base = 0
			modifier = {
				check_variable = { var = var:macedonian_arbitrators_favourite value = SER compare = equals }
				add = 100
			}
		}

		log = "Favouring Serbia in Macedonian arbitration."

		effect_tooltip = {
			every_state = {
				limit = {
					has_state_flag = macedonian_state_for_arbitration
				}
				transfer_state_to = SER
			}
		}

		set_variable = {
			var = global.favoured_in_macedonian_arbitration
			value = SER
		}

		SER = { country_event = wwi_serbia.14 }
		BUL = { country_event = wwi_serbia.15 }
	}

	# Rule in Bulgaria's favour.
	option = {
		name = wwi_serbia.13.b

		ai_chance = {
			base = 0
			modifier = {
				check_variable = { var = var:macedonian_arbitrators_favourite value = BUL compare = equals }
				add = 100
			}
		}

		log = "Favouring Bulgaria in Macedonian arbitration."

		effect_tooltip = {
			every_state = {
				limit = {
					has_state_flag = macedonian_state_for_arbitration
				}
				transfer_state_to = BUL
			}
		}

		set_variable = {
			var = global.favoured_in_macedonian_arbitration
			value = BUL
		}

		BUL = { country_event = wwi_serbia.14 }
		SER = { country_event = wwi_serbia.15 }
	}
}

# Event to determine macedonian territories on the Serbian-Bulgarian border after the Balkan War.
country_event = {
	id = wwi_serbia.16

	is_triggered_only = yes
	hidden = yes

	immediate = {
		# Identify all states in the macedonian area controlled by either Serbia or Bulgarian.
		every_state = {
			limit = {
				has_state_flag = disputed_macedonian_state
				is_owned_and_controlled_by = SER
			}
			
			set_state_flag = serbian_controlled_disputed_macedonian_state
			set_state_name = "Serbian Controlled Macedonia"
		}

		every_state = {
			limit = {
				has_state_flag = disputed_macedonian_state
				is_owned_and_controlled_by = BUL
			}
			
			set_state_flag = bulgarian_controlled_disputed_macedonian_state
			set_state_name = "Bulgarian Controlled Macedonia"
		}

		# Of these, mark any Serbia-controlled states bordering Bulgaria-controlled states as disputed, and vice versa.
		every_state = {
			limit = {
				has_state_flag = serbian_controlled_disputed_macedonian_state
			}
			every_neighbor_state = {
				limit = {
					has_state_flag = disputed_macedonian_state
					is_owned_and_controlled_by = BUL
				}

				set_state_flag = macedonian_state_for_arbitration
				set_state_name = "Border"
			}
		}

		every_state = {
			limit = {
				has_state_flag = bulgarian_controlled_disputed_macedonian_state
			}
			every_neighbor_state = {
				limit = {
					has_state_flag = disputed_macedonian_state
					is_owned_and_controlled_by = SER
				}

				set_state_flag = macedonian_state_for_arbitration
				set_state_name = "Border"
			}
		}
	}
}
