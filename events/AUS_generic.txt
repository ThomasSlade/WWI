country_event = {
	id = event_AUS_generic.4
	title = event_AUS_generic.4.t
	desc = {
		text = event_AUS_generic.4.d.a
		trigger = {
			check_variable = { SER.serb_vojvodina_armament < 2 }
		}
	}
	desc = {
		text = event_AUS_generic.4.d.b
		trigger = {
			check_variable = { SER.serb_vojvodina_armament = 2 }
		}
	}
	desc = {
		text = event_AUS_generic.4.d.c
		trigger = {
			check_variable = { SER.serb_vojvodina_armament > 2 }
		}
	}
	picture = GFX_report_event_german_speech

	is_triggered_only = yes
	trigger = {
		AND = {
			has_idea = AUS_vojvodina_separatists
			any_owned_state = { is_core_of = SVO }
		}
	}
	fire_only_once = yes

	mean_time_to_happen = {
		days = 1
	}

	immediate = {
		log = "([THIS.GetName] country event) Serbian Vojvodina declares independence."

		hidden_effect = {
			release = SVO
			set_global_flag = serb_vojvodina_revolted
			
			SER = {
				diplomatic_relation = {
					country = SVO
					relation = puppet
					active = yes
				}
			}

			SVO = {
				inherit_technology = SER

				if = {
					limit = { check_variable = { SER.serb_vojvodina_armament < 1 } }
					load_oob = "SVO_Strength_0"
				}
				else_if = {
					limit = { check_variable = { SER.serb_vojvodina_armament < 2 } }
					load_oob = "SVO_Strength_1"
				}
				else_if = {
					limit = { check_variable = { SER.serb_vojvodina_armament < 3 } }
					load_oob = "SVO_Strength_2"
				}
				else = {
					load_oob = "SVO_Strength_3"
				}
			}

			remove_ideas = AUS_vojvodina_separatists
		}
	}

	# Declare war on Vojvodina
	option = {
		name = event_AUS_generic.4.a
		ai_chance = {
			base = 1
		}

		declare_war_on = {
			target = SVO
			type = take_core_state
		}
	}

	# Let them go
	option = {
		trigger = {
			NOT = { has_war_with = SER }
		}

		name = event_AUS_generic.4.b
	}
}
