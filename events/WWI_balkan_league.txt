# The first Balkan League breaks out. This is the first time that a lot of countries will hear of the Balkan League (which was historically secret).

	id = event_WWI_balkan_league_first_balkan_war.1
	title = event_WWI_balkan_league_first_balkan_war.1.t
	desc = {
		text = event_WWI_balkan_league_first_balkan_war.1.d_turkey_humiliated
		trigger = {
			has_global_flag = TUR_humiliated_by_war
		}
	}
	desc = {
		text = event_WWI_balkan_league_first_balkan_war.1.d
	}
	picture = GFX_news_event_british_trench

	is_triggered_only = yes
	major = yes

	immediate = {
		log = "([THIS.GetName] news event) First Balkan War"
	}

	# Member.
	option = {
		name = event_WWI_balkan_league_first_balkan_war.1.a_member

		trigger = {
			has_country_flag = flag_balkan_coalition_member
		}
	}

	# Neutral.
	option = {
		name = event_WWI_balkan_league_first_balkan_war.1.a_neutral

		trigger = {
			NOT = { has_country_flag = flag_balkan_coalition_member }
			has_opinion = {
				target = FROM
				value > -20
			}
		}
	}

	# Negative.
	option = {
		name = event_WWI_balkan_league_first_balkan_war.1.a_negative

		trigger = {
			NOT = { has_country_flag = flag_balkan_coalition_member }

				target = FROM
				value < -19
			}
		}
	}
}

add_namespace = event_WWI_balkan_league_second_balkan_war

# The country with the most disputes will now need to declare the 2nd Balkan War.
country_event = {
	id = event_WWI_balkan_league_second_balkan_war.1
	title = event_WWI_balkan_league_second_balkan_war.1.t
	desc = event_WWI_balkan_league_second_balkan_war.1.d
	picture = GFX_report_event_german_speech

	is_triggered_only = yes

	immediate = {
		log = "([THIS.GetName] event) 2nd Balkan War Instigator must declare 2nd Balkan War."
		set_variable = { global.second_balkan_war_state = 1 }
	}

	# Okay.
	option = {
		name = event_WWI_balkan_league_second_balkan_war.1.a

		set_country_flag = flag_second_balkan_war_attacker
		set_variable = { global.second_balkan_war_attacker = THIS }

		leave_faction = yes

		# Have to do this absolutely insanely obtuse shit because is_in_faction doesn't actually fucking work with fucking variables.

		hidden_effect = {
			every_other_country = {
				limit = { has_country_flag = flag_balkan_coalition_member }

				ROOT = {
					create_wargoal = {
					    type = take_core_state
					    target = PREV
					}

					declare_war_on = {
		                target = PREV
		                type = take_core_state
		            }
				}

				set_country_flag = flag_second_balkan_war_defender
				country_event = { id = event_WWI_balkan_league_second_balkan_war.2 hours = 1 }
			}

			for_each_scope_loop = {
				array = neighbors
				if = {
					limit = {
						NOT = { has_country_flag = flag_second_balkan_war_defender }
						any_core_state = {
							is_controlled_by = ROOT
						}
					}
					country_event = { id = event_WWI_balkan_league_second_balkan_war.4 days = 10 }
				}
			}
		}
		

		effect_tooltip = {
            effect_SER_second_balkan_war_tt = yes
		}

		add_named_threat = {
			threat = 3
			name = threat_second_balkan_war_instigated
		}

		hidden_effect = {
			news_event = { id = event_WWI_balkan_league_second_balkan_war.3 days = 2 }
			var:global.balkan_coalition_founder = {
				country_event = { id = event_WWI_balkan_league_second_balkan_war.5 days = 3 }
			}
		}
	}
}

news_event = {
	id = event_WWI_balkan_league_second_balkan_war.6
	title = event_WWI_balkan_league_second_balkan_war.6.t
	desc = {
		text = event_WWI_balkan_league_second_balkan_war.6.d
		trigger = {
			check_variable = { global.second_balkan_war_state = 3 }
		}
	}
	desc = {
		text = event_WWI_balkan_league_second_balkan_war.6.d_attacker_wins
		trigger = {
			check_variable = { global.second_balkan_war_state = 4 }
		}
	}
	picture = GFX_news_event_british_trench

	is_triggered_only = yes
	major = yes

	immediate = {
		if = {
			limit = { check_variable = { global.second_balkan_war_state = 4 } }
			log = "([THIS.GetName] news event) 2nd Balkan War ends, instigator won."
		}
		else = {
			log = "([THIS.GetName] news event) 2nd Balkan War ends, instigator lost."
		}
	}
		

	# Loser.
	option = {
		name = event_WWI_balkan_league_second_balkan_war.6.a_loser

		trigger = {
			OR = {
				AND = {
					has_country_flag = flag_second_balkan_war_attacker
					check_variable = { global.second_balkan_war_state = 3 }
				}
				AND = {
					has_country_flag = flag_second_balkan_war_defender
					check_variable = { global.second_balkan_war_state = 4 }
				}
			}
		}
	}

	# Participant.
	option = {
		name = event_WWI_balkan_league_second_balkan_war.6.a_participant

		trigger = {
			OR = {
				AND = {
					has_country_flag = flag_second_balkan_war_attacker
					check_variable = { global.second_balkan_war_state = 4 }
				}
				AND = {
					has_country_flag = flag_second_balkan_war_defender
					check_variable = { global.second_balkan_war_state = 3 }
				}
			}
		}
	}

	# Neutral.
	option = {
		name = event_WWI_balkan_league_second_balkan_war.6.a_neutral

		trigger = {
			NOT = { has_country_flag = flag_second_balkan_war_defender }
			NOT = { has_country_flag = flag_second_balkan_war_attacker }
		}
	}
}

# Offer (force, really) a country to accept the resolution of a disputed state, in exchange for their choice of concession.
country_event = {
	id = event_WWI_balkan_league_resolve_dispute.1
	title = event_WWI_balkan_league_resolve_dispute.1.t
	desc = event_WWI_balkan_league_resolve_dispute.1.d
	picture = GFX_report_event_german_speech

	is_triggered_only = yes

	immediate = {
		log = "([THIS.GetName] event) Disputed Balkan state ([?disputed_state.GetName]) concession offer."
	}

	# Take a civ factory
	option = {
		name = event_WWI_balkan_league_resolve_dispute.1.a
		
		FROM = { effect_SER_balkan_concession_civ_give = yes }
		effect_SER_balkan_concession_civ_receive = yes

		1 = { save_event_target_as = concession_made }
		hidden_effect = {
			FROM = {
				country_event = { id = event_WWI_balkan_league_resolve_dispute.2 hours = 1 }
			}	
		}

		trigger = {
			FROM = { num_of_civilian_factories > 1 }
		}
	}

	# Take a mil factory
	option = {
		name = event_WWI_balkan_league_resolve_dispute.1.b

		FROM = { effect_SER_balkan_concession_mil_give = yes }
		effect_SER_balkan_concession_mil_receive = yes

		2 = { save_event_target_as = concession_made }
		hidden_effect = {
			FROM = {
				country_event = { id = event_WWI_balkan_league_resolve_dispute.2 hours = 1 }
			}	
		}

		trigger = {
			FROM = { num_of_military_factories > 1 }
		}
	}

	# Resource rights
	option = {
		name = event_WWI_balkan_league_resolve_dispute.1.c
		# TODO: Add resource rights after update

		3 = { save_event_target_as = concession_made }
		hidden_effect = {
			FROM = {
				country_event = { id = event_WWI_balkan_league_resolve_dispute.2 hours = 1 }
			}	
		}

		trigger = {
			event_target:disputed_state = {
				is_controlled_by = FROM
				trigger_has_any_resources = yes
			}
		}
	}
}
