add_namespace = event_GER_schlieffen_plan

country_event = {
	id = event_GER_schlieffen_plan.1
	title = event_GER_schlieffen_plan.1.t
	desc = {
		text = event_GER_schlieffen_plan.1.d_france
		trigger = { has_war_with = FRA }
	}
	desc = {
		text = event_GER_schlieffen_plan.1.d_generic
		trigger = { NOT = { has_war_with = FRA } }
	}
	picture = GFX_report_event_german_speech

	is_triggered_only = yes

	immediate = {
		log = "([THIS.GetName] event) Telling Germany they can do the Schlieffen Plan now."
	}

	# Invade all the targets
	# Ensure these effects stay in sync with the 'modifier_GER_schlieffen_plan_tt' tooltip.
	option = {
		name = event_GER_schlieffen_plan.1.a_france
		trigger = { has_war_with = FRA }
		ai_chance = { base = 100 }

		effect_GER_schlieffen_wargoals = yes

		declare_war_on = {
			target = BEL
			type = puppet_wargoal_focus
		}

		declare_war_on = {
			target = LUX
			type = annex_everything
		}

		if = {
			limit = { has_completed_focus = GER_schlieffen_unaltered }
			declare_war_on = {
				target = HOL
				type = puppet_wargoal_focus
			}
		}
	}

	option = {
		name = event_GER_schlieffen_plan.1.a_generic
		trigger = { NOT = { has_war_with = FRA } }
		ai_chance = { base = 100 }

		effect_GER_schlieffen_wargoals = yes

		declare_war_on = {
			target = BEL
			type = puppet_wargoal_focus
		}

		declare_war_on = {
			target = LUX
			type = annex_everything
		}

		if = {
			limit = { has_completed_focus = GER_schlieffen_unaltered }
			declare_war_on = {
				target = HOL
				type = puppet_wargoal_focus
			}
		}
	}

	# Invade the LUX and BEL but not HOL
	option = {
		name = event_GER_schlieffen_plan.1.b
		trigger = { has_completed_focus = GER_schlieffen_unaltered }
		ai_chance = { base = 0 }

		effect_GER_schlieffen_wargoals = yes

		declare_war_on = {
			target = BEL
			type = puppet_wargoal_focus
		}

		declare_war_on = {
			target = LUX
			type = annex_everything
		}
	}

	# Don't invade
	option = {
		name = event_GER_schlieffen_plan.1.c
		ai_chance = { base = 0 }

		effect_GER_schlieffen_wargoals = yes
	}
}
