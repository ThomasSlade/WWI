news_event = {
	id = event_food_shortage.5
	title = {
		text = event_food_shortage.5.t_turnip_winter
		trigger = {
			trigger_is_spring_or_summer = no
			NOT = { has_global_flag = flag_turnip_winter_happened }
		}
	}
	title = {
		text = event_food_shortage.5.t
	}
	desc = {
		text = event_food_shortage.5.d_no_blockade
		trigger = {
			any_enemy_country = {
				is_major = yes
				NOT = { has_idea = open_sealanes }
			}
		}
	}
	desc = {
		text = event_food_shortage.5.d_blockade
	}
	picture = GFX_news_event_british_trench

	is_triggered_only = yes
	major = yes

	immediate = {
		log = "([THIS.GetName] news event) Starvation news event."
		set_global_flag = { flag = flag_starvation_news_event_happened value = 1 days = 60 }

		if = {
			trigger = {
				trigger_is_spring_or_summer = no
				NOT = { has_global_flag = flag_turnip_winter_happened }
			}
			set_global_flag = flag_turnip_winter_happened
		}
	}

	option = {
		name = event_food_shortage.5.a_neutral
		trigger = {
			NOT = { has_war_with = FROM }
		}
	}

	option = {
		name = event_food_shortage.5.a_negative
		trigger = {
			has_war_with = FROM
		}
	}
}
