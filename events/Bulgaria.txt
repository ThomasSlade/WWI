add_namespace = wwi_bulgaria

# Offer to Join Balkan League
country_event = {
	id = wwi_bulgaria.1
	title = wwi_bulgaria.1.t
	desc = {
		trigger = { SER = { has_completed_focus = SER_seek_russian_mediation } }
		text = wwi_bulgaria.2.d_russia
	}
	desc = {
		trigger = { SER = { has_completed_focus = SER_appeal_to_austria_for_support } }
		text = wwi_bulgaria.1.d_austria
	}
	picture = GFX_report_event_german_speech

	is_triggered_only = yes

	# Accept
	option = {
		name = wwi_bulgaria.1.a
		ai_chance = { factor = 1 }
		
		# Give non-Macedonian cores to Bulgaria (includes Salonika).
		add_state_core = 157
		add_state_core = 220
		add_state_core = 133
		add_state_core = 217
		add_state_core = 226
		add_state_core = 127
		add_state_core = 188
		add_state_core = 140

		if = {
			limit = { AUS = { has_country_flag = macedonian_arbitrator } }
			custom_effect_tooltip = wwi_macedonian_arbitration_austria_tt
		}
		else = {
			custom_effect_tooltip = wwi_macedonian_arbitration_russia_tt
		}

		hidden_effect = {
			FROM = { country_event = wwi_serbia.3 }
		}
	}

	# Decline
	option = {
		name = wwi_bulgaria.1.b
		ai_chance = {
			factor = 0
			modifier = {
				add = 5
				strength_ratio = {
					tag = FROM
					ratio > 2
				}
			}
		}
	}
}
