add_namespace = event_SOV_russian_reform

country_event = {
	id = event_SOV_russian_reform.1
	title = event_SOV_russian_reform.1.t
	desc = event_SOV_russian_reform.1.d
	picture = GFX_report_event_german_speech

	is_triggered_only = yes

	immediate = {
		log = "([THIS.GetName] event) Russian city was land-reformed."
	}

	# Assure the Tsar, get Tsar favour
	option = {
		name = event_SOV_russian_reform.1.a
		effect_SOV_gain_2_tsar_favor = yes
	}

	# Continue industrialisation, get factories
	option = {
		name = event_SOV_russian_reform.1.b
		event_target:reformed_state = {
			add_extra_state_shared_building_slots = 1
			add_building_construction = {
				type = industrial_complex
				level = 1
				instant_build = yes
			}

			add_extra_state_shared_building_slots = 1
			add_building_construction = {
				type = arms_factory
				level = 1
				instant_build = yes
			}
		}
	}

	# Build infrastructure
	option = {
		name = event_SOV_russian_reform.1.c
		event_target:reformed_state = {
			every_neighbor_state = {
				limit = { is_owned_by = ROOT }
				add_building_construction = {
					type = infrastructure
					level = 1
					instant_build = yes
				}
			}
		}
	}

	# Gain more resources
	option = {
		name = event_SOV_russian_reform.1.e
		event_target:reformed_state = {
			every_neighbor_state = {
				limit = { is_owned_by = ROOT }
				add_dynamic_modifier = {
					modifier = modifier_SOV_exploited_resources
				}
			}
		}
	}
}
