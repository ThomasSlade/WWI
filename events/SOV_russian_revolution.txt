add_namespace = event_SOV_russian_revolution

country_event = {
	id = event_SOV_russian_revolution.1
	title = event_SOV_russian_revolution.1.t
	desc = {
		trigger = {
			has_idea = idea_SOV_autocratic_tsardom
		}
		text = event_SOV_russian_revolution.1.d
	}
	desc = {
		trigger = {
			NOT = { has_idea = idea_SOV_autocratic_tsardom }
		}
		text = event_SOV_russian_revolution.1.d_not_tsarist
	}
	picture = GFX_report_event_german_speech

	is_triggered_only = yes

	immediate = {
		log = "([THIS.GetName] event) Warning of impending Civil War and starting mission."
		hidden_effect = {
			activate_mission = mission_SOV_bolshevik_revolution_imminent

			if = {
				limit = { NOT = { has_country_flag = flag_SOV_revolution_warning_news_fired } }
				news_event = {
					id = event_SOV_russian_revolution.3
					days = 2
				}
				set_country_flag = flag_SOV_revolution_warning_news_fired
			}
		}
	}

	option = {
		trigger = {
			has_idea = idea_SOV_autocratic_tsardom
		}
		name = event_SOV_russian_revolution.1.a

		effect_tooltip = {
			activate_mission = mission_SOV_bolshevik_revolution_imminent
		}
	}

	option = {
		trigger = {
			NOT = { has_idea = idea_SOV_autocratic_tsardom }
		}
		name = event_SOV_russian_revolution.1.a_not_tsarist

		effect_tooltip = {
			activate_mission = mission_SOV_bolshevik_revolution_imminent
		}
	}
}
