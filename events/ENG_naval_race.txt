add_namespace = event_ENG_naval_race

country_event = {
	id = event_ENG_naval_race.1
	title = event_ENG_naval_race.1.t
	desc = {
		text = event_ENG_naval_race.1.d_britain
		trigger = {
			original_tag = ENG
		}
	}
	desc = {
		text = event_ENG_naval_race.1.d_rival
		trigger = {
			NOT = { original_tag = ENG }
		}
	}
	picture = GFX_report_event_german_speech

	is_triggered_only = yes

	immediate = {
		log = "([THIS.GetName] event) Britain takes lead in naval race."
	}

	option = {
		name = event_ENG_naval_race.1.a_britain
		trigger = {
			original_tag = ENG
		}

		effect_tooltip = {
			set_temp_variable = { param_naval_race_idea_britain = token:idea_ENG_naval_race_britain_0_british_bonus }
			set_temp_variable = { param_naval_race_idea_rival = token:idea_ENG_naval_race_britain_0_rival_malus }
			effect_ENG_set_naval_race_ideas = yes
		}
	}

	option = {
		name = event_ENG_naval_race.1.a_rival
		trigger = {
			NOT = { original_tag = ENG }
		}
	}
}
