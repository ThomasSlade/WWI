add_namespace = event_wwi_alignment

country_event = {
	id = event_wwi_alignment.1
	title = event_wwi_alignment.1.t
	desc = event_wwi_alignment.1.d
	picture = GFX_report_event_german_speech

	is_triggered_only = yes

	immediate = {
		log = "([THIS.GetName] event) Has chance to adopt new alignment ideology."
	}

	option = {
		name = event_wwi_alignment.1.a

		set_temp_variable = { param_alignment = event_target:alignment_target.leader_of_alignment }
		effect_set_politics_token_friendly = yes
	}

	option = {
		name = event_wwi_alignment.1.b

		custom_effect_tooltip = effect_can_change_alignment_later_tt
	}
}

news_event = {
	id = event_wwi_become_great_power.3
	title = event_wwi_become_great_power.3.t
	desc = event_wwi_become_great_power.3.d
	picture = GFX_news_event_british_trench

	is_triggered_only = yes
	major = yes

	immediate = {
		log = "([THIS.GetName] news event) Informing that [FROM] has become a great power and is using a custom influence colour."
	}

	option = {
		name = event_wwi_become_great_power.3.a_self
		trigger = {
			tag = FROM
		}

		custom_effect_tooltip = effect_from_became_great_power_tt
	}

	option = {
		name = event_wwi_become_great_power.3.a_neutral
		trigger = {
			NOT = { tag = FROM }
			NOT = { has_war_with = FROM }
		}

		custom_effect_tooltip = effect_from_became_great_power_tt
	}

	option = {
		name = event_wwi_become_great_power.3.a_negative
		trigger = {
			NOT = { tag = FROM }
			has_war_with = FROM
		}

	}
