	id = event_AUS_reform.1
	title = event_AUS_reform.1.t
	desc = event_AUS_reform.1.d
	picture = GFX_news_event_british_trench

	is_triggered_only = yes
	major = yes

	immediate = {
		log = "([THIS.GetName] news event) United States of Greater Austria is formed."
	}

	# Austria
	option = {
		name = event_AUS_reform.1.a

		trigger = {
			original_tag = AUS
		}
	}

	# Hungary
	option = {
		name = event_AUS_reform.1.d
		effect_tooltip = {
			FROM = { AUS_form_united_states_of_greater_austria = yes }
		}

		trigger = {
			original_tag = HUN
		}
	}

	# Angry
	option = {
		name = event_AUS_reform.1.c
		effect_tooltip = {
			FROM = { AUS_form_united_states_of_greater_austria = yes }
		}

		trigger = {
			NOT = { original_tag = AUS }
			NOT = { original_tag = HUN }
			OR = {
				has_opinion = {
					target = FROM
					value < -49
				}
				has_war_with = FROM
			}
		}
	}

	# Indifferent
	option = {
		name = event_AUS_reform.1.b
		effect_tooltip = {
			FROM = { AUS_form_united_states_of_greater_austria = yes }
		}

		trigger = {
			NOT = { original_tag = AUS }
			NOT = { original_tag = HUN }
			AND = {
				has_opinion = {
					target = FROM
					value > -50
				}
				NOT = { has_war_with = FROM }
			}
		}
	}
}

	is_triggered_only = yes

	immediate = {
		log = "([THIS.GetName] country event) Get offered Bukovina by Austria."
	}
	
	option = {
		name = event_AUS_reform.8.a
		ai_chance = { base = 1 }

		transfer_state = 236
		add_timed_idea = { idea = AUS_trade_for_bukovina days = 730 }
		add_opinion_modifier = {
			target = FROM
			modifier = AUS_received_bukovina
		}
		FROM = {
			add_opinion_modifier = {
				target = PREV
				modifier = AUS_gave_bukovina
			}
		}
	}

	option = {
		name = event_AUS_reform.8.b
		ai_chance = { base = 0 }
	}
}
