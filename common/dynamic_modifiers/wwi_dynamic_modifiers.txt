	enable = { always = yes }

		OR = {
			# Remove if not held by Austria, or if a core of the target nations but none of the required advisors are employed.
			controller = {
				NOT = { original_tag = AUS }
			}
			# None of the following can be true if the effect is to be removed.
			NOT = {
				OR = {
					AND = {
						is_core_of = CZE

					}
					AND = {
						is_core_of = SCS

						is_core_of = UKR

			}
