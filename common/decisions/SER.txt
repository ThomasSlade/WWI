	# Annex Bosnia after they've finalised their secession: This is available if Serbia doesn't want to immediately annex Bosnia, e.g. in the event that Bosnia is surrounded by hostile territory.
	SER_accept_bosnian_secession = {
		allowed = {
			original_tag = SER
		}

		available = {

		cost = 0
		fire_only_once = yes

		ai_will_do = {
			factor = 1
		}

		visible = {
			AND = {
				# Hide the event unless the target isn't Serbia.
				NOT = { FROM = { original_tag = SER } }
				OR = {
					FROM = { owns_state = 224 }
					FROM = { owns_state = 146 }
					FROM = { owns_state = 169 }
				}
			}
		}

		complete_effect = {
			if = {
				limit = { FROM = { owns_state = 224 } }

			}
			if = {
				limit = { FROM = { owns_state = 146 } }
				transfer_state = 146
				add_state_core = 146
			}
			if = {
				limit = { FROM = { owns_state = 169 } }
				transfer_state = 169
				add_state_core = 169
			}
			add_timed_idea = {
				idea = territory_restored
				days = 180
			}

			hidden_effect = {
				FROM = { country_event = wwi_austria.8 }
				FROM = {
					remove_ideas = {
						AUS_narodna_odbrana
						AUS_black_hand_cells
					}
				}
			}
		}
	}
