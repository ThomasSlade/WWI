AUS_influence_adriatic_territory = {
	AUS_influence_free_city = {
		targets = { TRS ZAR }

		available = {
			has_navy_experience > 99
			if = {
				limit = { original_tag = AUS }
				FROM = { NOT = { has_idea = AUS_austrian_influenced_free_city } }
			}
			if = {
				limit = { original_tag = ITA }
				FROM = { NOT = { has_idea = AUS_italian_influenced_free_city } }
			}
		}

		complete_effect = {
			navy_experience = -100
			if = {
				limit = { original_tag = AUS }
				FROM = { AUS_sway_free_city_to_austria = yes }
			}
			else_if = {
				limit = { original_tag = ITA }
				FROM = { AUS_sway_free_city_to_italy = yes }
			}

			hidden_effect = {
				FROM = { save_event_target_as = influenced_free_city }
				if = {
					limit = { original_tag = AUS }
					ITA = { country_event = { id = wwi_austria.17 hours = 1 } }
				}
				else_if = {
					limit = { original_tag = ITA }
					AUS = { country_event = { id = wwi_austria.17 hours = 1 } }
				}
			}
		}

		ai_will_do = {
			base = 0
			modifier = {
				add = 1
				has_navy_experience > 149
			}
			modifier = {
				add = 1
				is_neighbor_of = FROM
			}
			# If you're already close to having one free city annexed, deprioritize spending experience on another one.
			modifier = {
				factor = 0
				if = {
					limit = { tag = AUS }
					OR = {
						TRS = { has_idea = AUS_austrian_influenced_free_city }
						ZAR = { has_idea = AUS_austrian_influenced_free_city }
					}
				}
				else = {
					OR = {
						TRS = { has_idea = AUS_italian_influenced_free_city }
						ZAR = { has_idea = AUS_italian_influenced_free_city }
					}
				}
			}
			# If this is the AI's favoured free city, but the opponent has influenced it, work extra hard to undo the influence.
			modifier = {
				add = 1
				AND = {
					check_variable = { priority_free_city = FROM }
					if = {
						limit = { tag = AUS }
						FROM = { has_idea = AUS_italian_influenced_free_city }
					}
					else = {
						FROM = { has_idea = AUS_austrian_influenced_free_city }
					}
				}
			}
		}
	}

	AUS_annex_free_ciy = {
		targets = { TRS ZAR }

		available = {
			has_navy_experience > 199
			if = {
				limit = { original_tag = AUS }
				FROM = { has_idea = AUS_austrian_influenced_free_city }
			}
			if = {
				limit = { original_tag = ITA }
				FROM = { has_idea = AUS_italian_influenced_free_city }
			}
		}

		complete_effect = {
			navy_experience = -200
			annex_country = { target = FROM transfer_troops = no }
			hidden_effect = {
				remove_ideas = { AUS_austrian_influenced_free_city AUS_italian_influenced_free_city }
				FROM = {
					save_event_target_as = influenced_free_city
					# Important that these get removed. Otherwise, they remain on the dead countries and it messes with the AI's judgement.
					remove_ideas = { AUS_austrian_influenced_free_city AUS_italian_influenced_free_city }
				}
				if = {
					limit = { original_tag = AUS }
					ITA = { country_event = { id = wwi_austria.18 hours = 1 } }
				}
				else_if = {
					limit = { original_tag = ITA }
					AUS = { country_event = { id = wwi_austria.18 hours = 1 } }
				}
			}
		}

		ai_will_do = {
			base = 0
			modifier = {
				add = 1
				has_navy_experience > 249
			}
			modifier = {
				add = 1
				is_neighbor_of = FROM
			}
			modifier = {
				factor = 3
				check_variable = { priority_free_city = FROM }
			}
		}
	}
}
