@PACIFY_CITY_TARGET_MIL_WEIGHT = 0.5 # Each mil factory matters by this factor when the AI is selecting a pacify_city target

# If a neighboring state has this resource (in any quantity), this value is added when evaluating it for pacify_city target.
# WWI TODO: Rename these when resources are changed.

@PACIFY_CITY_TARGET_ALUMINIUM_WEIGHT = 2
@PACIFY_CITY_TARGET_RUBBER_WEIGHT = 1
@PACIFY_CITY_TARGET_TUNGSTEN_WEIGHT = 1
@PACIFY_CITY_TARGET_STEEL_WEIGHT = 3
@PACIFY_CITY_TARGET_CHROMIUM_WEIGHT = 1

@PACIFY_CITY_TARGET_REFORMED_NEIGHBORS_WEIGHT = 3 # Add this much for each of the neighbors of the city that have been reformed, making it cheaper
@PACIFY_CITY_TARGET_DISCOUNTED_NEIGHBORS_WEIGHT = 2 # Add this much for each of the neighbors of the city that have the discount, making it cheaper

	# This decision is used by the AI to set the state it wants to focus on to unlock pacify_city, so that the AI focuses on reforming land surrounding this city.
	decision_ai_select_pacify_city_target = {
		state_target = yes
		on_map_mode = map_only

		visible = {
			is_ai = yes
		}

		target_trigger = {
			FROM = {
				is_owned_and_controlled_by = ROOT
				has_dynamic_modifier = {
					modifier = modifier_SOV_unreformed_city
				}
			}
		}

		complete_effect = {
			set_variable = { pacify_city_target = FROM }
			log = "([THIS.GetName] Land Reform) AI has selected target to focus on unlocking pacify_city decision: [?pacify_city_target.GetName]"
		}

		ai_will_do = {
			base = 1
			modifier = {
				# If there's already a pacify city target, and it's still valid, don't pick a new one.
				factor = 0
				AND = {
					has_variable = pacify_city_target
					var:pacify_city_target = {
						AND = {
							is_owned_and_controlled_by = ROOT
							has_dynamic_modifier = {
								modifier = modifier_SOV_unreformed_city
							}
						}
					}
				}
			}

			modifier = {
				set_temp_variable = { civ_score = building_level@industrial_complex }
				multiply_temp_variable = { civ_score = @PACIFY_CITY_TARGET_CIV_WEIGHT }
				add = civ_score
			}

			modifier = {
				set_temp_variable = { mil_score = building_level@arms_factory }
				multiply_temp_variable = { mil_score = @PACIFY_CITY_TARGET_MIL_WEIGHT }
				add = mil_score
			}
			
			# Each variety of resource in each neighboring state adds value. I don't go to the trouble of evaluating them proportional to the resource quantity.
			modifier = {
				set_temp_variable = { resource_score = 0 }
				FROM = {
					all_neighbor_state = {
						if = {
							limit = {
								oil > 0
							}

							add_to_temp_variable = { resource_score = @PACIFY_CITY_TARGET_OIL_WEIGHT }
						}
						if = {
							limit = {
								aluminium > 0
							}

							add_to_temp_variable = { resource_score = @PACIFY_CITY_TARGET_ALUMINIUM_WEIGHT }
						}
						if = {
							limit = {
								rubber > 0
							}

							add_to_temp_variable = { resource_score = @PACIFY_CITY_TARGET_RUBBER_WEIGHT }
						}
						if = {
							limit = {
								tungsten > 0
							}

							add_to_temp_variable = { resource_score = @PACIFY_CITY_TARGET_TUNGSTEN_WEIGHT }
						}
						if = {
							limit = {
								steel > 0
							}

							add_to_temp_variable = { resource_score = @PACIFY_CITY_TARGET_STEEL_WEIGHT }
						}
						if = {
							limit = {
								chromium > 0
							}

							add_to_temp_variable = { resource_score = @PACIFY_CITY_TARGET_CHROMIUM_WEIGHT }
						}

						if = {
							limit = {
								trigger_SOV_is_unreformed_state = no
							}

							add_to_temp_variable = { resource_score = @PACIFY_CITY_TARGET_REFORMED_NEIGHBORS_WEIGHT }
						}
						else_if = {
							limit = {
								any_neighbor_state = {
									trigger_SOV_is_unreformed_state = no
								}
							}

							add_to_temp_variable = { resource_score = @PACIFY_CITY_TARGET_DISCOUNTED_NEIGHBORS_WEIGHT }
						}
					}
				}

				add = resource_score
			}
		}
	}
