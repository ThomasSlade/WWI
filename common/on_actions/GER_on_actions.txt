on_actions = {
	#ROOT = attacking side
	#FROM = defending side
	on_war_relation_added = {
		effect = {
			if = {
				limit = {
					OR = {
						ROOT = { original_tag = GER }
						FROM = { original_tag = GER }
					}
				}
			}
			if = {
				limit = { ROOT = { has_idea = idea_GER_military_travel_plan_primer } }
				remove_ideas = idea_GER_military_travel_plan_primer
			}
			else_if = {
				limit = { FROM = { has_idea = idea_GER_military_travel_plan_primer } }
				FROM  = { remove_ideas = idea_GER_military_travel_plan_primer }
			}

			if = {
				limit = {
					has_idea = idea_GER_schlieffen_plan
					OR = {
						ROOT = { original_tag = FRA }
						FROM = { original_tag = FRA }
						ROOT = { original_tag = ENG }
						FROM = { original_tag = ENG }
					}
				}

				effect_GER_enable_schlieffen_plan = yes
			}
		}
	}

	on_startup = {
		effect = {
			GER = {
				effect_GER_init_staatenbund_industry_share = yes
			}
		}
	}

	on_weekly_GER = {
		effect = {
			effect_GER_update_staatenbund_industry_share = yes
		}
	}
}
