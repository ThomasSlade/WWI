@DREADNOUGHT_RATIO_RIVAL_LEAD_0 = 1.35	# If the ENG-to-rival dreadnought ratio is less than this, the rival has a weak lead.

@HEAVY_RATIO_RIVAL_LEAD_1 = 1.2
@HEAVY_RATIO_ENG_LEAD_0 = 1.5
@HEAVY_RATIO_ENG_LEAD_1 = 2.0

					}
					else = {
						set_variable = { britain_to_rival_dreadnoughts = 1000 }
					}
				}
				else = {
					set_variable = { britain_to_rival_dreadnoughts = britain_dreadnoughts }

				add_to_temp_variable = { britain_heavies = num_ships_with_type@carrier }
				add_to_temp_variable = { britain_heavies = num_ships_with_type@pre_dreadnought_wwi }
				var:naval_race_rival = {
					set_temp_variable = { rival_heavies = num_ships_with_type@battleship }
					add_to_temp_variable = { rival_heavies = num_ships_with_type@heavy_cruiser }
					add_to_temp_variable = { rival_heavies = num_ships_with_type@carrier }
					add_to_temp_variable = { rival_heavies = num_ships_with_type@pre_dreadnought_wwi }
				}

				if = {
					limit = { check_variable = { rival_heavies = 0 } }
					if = {
						limit = { check_variable = { britain_heavies = 0 } }
						set_variable = { britain_to_rival_heavies = 1 }
					}
					else = {
						set_variable = { britain_to_rival_heavies = 1000 }
					}
				}
				else = {
					set_variable = { britain_to_rival_heavies = britain_heavies }
					divide_variable = { britain_to_rival_heavies = rival_heavies }
				}

				### Get light ratio
				set_temp_variable = { britain_lights = num_ships_with_type@light_cruiser }
				add_to_temp_variable = { britain_lights = num_ships_with_type@destroyer }
				var:naval_race_rival = {

					set_temp_variable = { rival_lights = num_ships_with_type@light_cruiser }
					add_to_temp_variable = { rival_lights = num_ships_with_type@destroyer }
				}

				if = {
					limit = { check_variable = { rival_lights = 0 } }
					if = {
						limit = { check_variable = { britain_lights = 0 } }
						set_variable = { britain_to_rival_lights = 1 }
					}
					else = {
						set_variable = { britain_to_rival_lights = 1000 }
					}
				}
				else = {
					set_variable = { britain_to_rival_lights = britain_lights }
					divide_variable = { britain_to_rival_lights = rival_lights }
				}

				### Get submarine ratio
				set_temp_variable = { britain_submarines = num_ships_with_type@submarine }
				var:naval_race_rival = {
					set_temp_variable = { rival_submarines = num_ships_with_type@submarine }
				}

				if = {
					limit = { check_variable = { rival_submarines = 0 } }
					if = {
						limit = { check_variable = { britain_submarines = 0 } }
						set_variable = { britain_to_rival_submarines = 1 }
					}
					else = {
						set_variable = { britain_to_rival_submarines = 1000 }
					}
				}
				else = {
					set_variable = { britain_to_rival_submarines = britain_submarines }
					divide_variable = { britain_to_rival_submarines = rival_submarines }
				}

				### Dreadnoughts
				if = {
					limit = {
						AND = {
							check_variable = { britain_to_rival_dreadnoughts < @DREADNOUGHT_RATIO_RIVAL_LEAD_0 }
							check_variable = { var = britain_to_rival_dreadnoughts value = @DREADNOUGHT_RATIO_RIVAL_LEAD_1 compare = greater_than_or_equals }
						}
					}

					add_power_balance_modifier = { id = bop_ENG_naval_race modifier = rival_more_dreadnoughts_0 }
				}
				else = {
					remove_power_balance_modifier = { id = bop_ENG_naval_race modifier = rival_more_dreadnoughts_0 }
				}

				if = {
					limit = { check_variable = { britain_to_rival_dreadnoughts < @DREADNOUGHT_RATIO_RIVAL_LEAD_1 } }

					add_power_balance_modifier = { id = bop_ENG_naval_race modifier = rival_more_dreadnoughts_1 }
				}
				else = {
					remove_power_balance_modifier = { id = bop_ENG_naval_race modifier = rival_more_dreadnoughts_1 }
				}

				if = {
					limit = {
						AND = {
							check_variable = { britain_to_rival_dreadnoughts > @DREADNOUGHT_RATIO_ENG_LEAD_0 }
							check_variable = { var = britain_to_rival_dreadnoughts value = @DREADNOUGHT_RATIO_ENG_LEAD_1 compare = less_than_or_equals }
						}
					}

					add_power_balance_modifier = { id = bop_ENG_naval_race modifier = britain_more_dreadnoughts_0 }
				}
				else = {
					remove_power_balance_modifier = { id = bop_ENG_naval_race modifier = britain_more_dreadnoughts_0 }
				}

				if = {
					limit = { check_variable = { britain_to_rival_dreadnoughts > @DREADNOUGHT_RATIO_ENG_LEAD_1 } }

					add_power_balance_modifier = { id = bop_ENG_naval_race modifier = britain_more_dreadnoughts_1 }
				}
				else = {
					remove_power_balance_modifier = { id = bop_ENG_naval_race modifier = britain_more_dreadnoughts_1 }
				}

				### Heavies
				if = {
					limit = {
						AND = {
							check_variable = { britain_to_rival_heavies < @HEAVY_RATIO_RIVAL_LEAD_0 }
							check_variable = { var = britain_to_rival_heavies value = @HEAVY_RATIO_RIVAL_LEAD_1 compare = greater_than_or_equals }
						}
					}

					add_power_balance_modifier = { id = bop_ENG_naval_race modifier = rival_more_heavies_0 }
				}
				else = {
					remove_power_balance_modifier = { id = bop_ENG_naval_race modifier = rival_more_heavies_0 }
				}

				if = {
					limit = { check_variable = { britain_to_rival_heavies < @HEAVY_RATIO_RIVAL_LEAD_1 } }

					add_power_balance_modifier = { id = bop_ENG_naval_race modifier = rival_more_heavies_1 }
				}
				else = {
					remove_power_balance_modifier = { id = bop_ENG_naval_race modifier = rival_more_heavies_1 }
				}

				if = {
					limit = {
						AND = {
							check_variable = { britain_to_rival_heavies > @HEAVY_RATIO_ENG_LEAD_0 }
							check_variable = { var = britain_to_rival_heavies value = @HEAVY_RATIO_ENG_LEAD_1 compare = less_than_or_equals }
						}
					}

					add_power_balance_modifier = { id = bop_ENG_naval_race modifier = britain_more_heavies_0 }
				}
				else = {
					remove_power_balance_modifier = { id = bop_ENG_naval_race modifier = britain_more_heavies_0 }
				}

				if = {
					limit = { check_variable = { britain_to_rival_heavies > @HEAVY_RATIO_ENG_LEAD_1 } }

					add_power_balance_modifier = { id = bop_ENG_naval_race modifier = britain_more_heavies_1 }
				}
				else = {
					remove_power_balance_modifier = { id = bop_ENG_naval_race modifier = britain_more_heavies_1 }
				}

				### Lights
				if = {
					limit = {
						AND = {
							check_variable = { britain_to_rival_lights < @LIGHT_RATIO_RIVAL_LEAD_0 }
							check_variable = { var = britain_to_rival_lights value = @LIGHT_RATIO_RIVAL_LEAD_1 compare = greater_than_or_equals }
						}
					}

					add_power_balance_modifier = { id = bop_ENG_naval_race modifier = rival_more_lights_0 }
				}
				else = {
					remove_power_balance_modifier = { id = bop_ENG_naval_race modifier = rival_more_lights_0 }
				}

				if = {
					limit = { check_variable = { britain_to_rival_lights < @LIGHT_RATIO_RIVAL_LEAD_1 } }

					add_power_balance_modifier = { id = bop_ENG_naval_race modifier = rival_more_lights_1 }
				}
				else = {
					remove_power_balance_modifier = { id = bop_ENG_naval_race modifier = rival_more_lights_1 }
				}

				if = {
					limit = {
						AND = {
							check_variable = { britain_to_rival_lights > @LIGHT_RATIO_ENG_LEAD_0 }
							check_variable = { var = britain_to_rival_lights value = @LIGHT_RATIO_ENG_LEAD_1 compare = less_than_or_equals }
						}
					}

					add_power_balance_modifier = { id = bop_ENG_naval_race modifier = britain_more_lights_0 }
				}
				else = {
					remove_power_balance_modifier = { id = bop_ENG_naval_race modifier = britain_more_lights_0 }
				}

				if = {
					limit = { check_variable = { britain_to_rival_lights > @LIGHT_RATIO_ENG_LEAD_1 } }

					add_power_balance_modifier = { id = bop_ENG_naval_race modifier = britain_more_lights_1 }
				}
				else = {
					remove_power_balance_modifier = { id = bop_ENG_naval_race modifier = britain_more_lights_1 }
				}

				### Submarines
				if = {
					limit = {
						AND = {
							check_variable = { britain_to_rival_submarines < @SUBMARINE_RATIO_RIVAL_LEAD_0 }
							check_variable = { var = britain_to_rival_submarines value = @SUBMARINE_RATIO_RIVAL_LEAD_1 compare = greater_than_or_equals }
						}
					}

					add_power_balance_modifier = { id = bop_ENG_naval_race modifier = rival_more_submarines_0 }
				}
				else = {
					remove_power_balance_modifier = { id = bop_ENG_naval_race modifier = rival_more_submarines_0 }
				}

				if = {
					limit = { check_variable = { britain_to_rival_submarines < @SUBMARINE_RATIO_RIVAL_LEAD_1 } }

					add_power_balance_modifier = { id = bop_ENG_naval_race modifier = rival_more_submarines_1 }
				}
				else = {
					remove_power_balance_modifier = { id = bop_ENG_naval_race modifier = rival_more_submarines_1 }
				}

				if = {
					limit = {
						AND = {
							check_variable = { britain_to_rival_submarines > @SUBMARINE_RATIO_ENG_LEAD_0 }
							check_variable = { var = britain_to_rival_submarines value = @SUBMARINE_RATIO_ENG_LEAD_1 compare = less_than_or_equals }
						}
					}

					add_power_balance_modifier = { id = bop_ENG_naval_race modifier = britain_more_submarines_0 }
				}
				else = {
					remove_power_balance_modifier = { id = bop_ENG_naval_race modifier = britain_more_submarines_0 }
				}

				if = {
					limit = { check_variable = { britain_to_rival_submarines > @SUBMARINE_RATIO_ENG_LEAD_1 } }

					add_power_balance_modifier = { id = bop_ENG_naval_race modifier = britain_more_submarines_1 }
				}
				else = {
					remove_power_balance_modifier = { id = bop_ENG_naval_race modifier = britain_more_submarines_1 }
				}
			}
		}
	}
}
