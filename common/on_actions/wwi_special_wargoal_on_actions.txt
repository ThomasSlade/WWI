on_actions = {
	on_daily_ITA = {
		effect = {
			if = {
				limit = {
					has_country_flag = flag_italo_turkish_war
					NOT = {
						has_country_flag = {
							flag = flag_italo_turkish_war
							value > 1
						}
					}
					trigger_special_wargoal_italo_turkish = yes
				}

				country_event = { id = event_special_wargoal_italo_turkish.1 }
			}
		}
	}

	on_daily_SER = {
		effect = {
			if = {
				limit = {
					has_country_flag = flag_first_balkan_war
					NOT = {
						has_country_flag = {
							flag = flag_first_balkan_war
							value > 1
						}
					}
					trigger_special_wargoal_first_balkan = yes
				}

				country_event = { id = event_special_wargoal_first_balkan.1 }
			}
		}
	}

	on_peaceconference_started = {
		effect = {
			if = {
				limit = {
					trigger_special_wargoal_italo_turkish_peace = yes
				}

				log = "Removing the flag_italo_turkish_war flag in peace conference between [FROM.GetName] and [ROOT.GetName]."

				ROOT = { clr_country_flag = flag_italo_turkish_war }
				FROM = { clr_country_flag = flag_italo_turkish_war }
			}

			if = {
				limit = {
					trigger_special_wargoal_first_balkan_peace = yes
				}

				log = "Removing the flag_first_balkan_war flag in peace conference between [FROM.GetName] and [ROOT.GetName]."

				ROOT = { clr_country_flag = flag_first_balkan_war }
				FROM = { clr_country_flag = flag_first_balkan_war }
			}
		}
	}
}
