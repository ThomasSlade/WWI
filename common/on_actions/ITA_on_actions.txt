on_actions = {
	on_daily_ITA = {
		effect = {
			if = {
				limit = {
					NOT = { has_country_flag = flag_ITA_conquer_libya_event_has_fired }
					trigger_ITA_conquer_libya = yes
				}

				country_event = { id = event_ITA_italo_turkish_war.1 }
			}
		}
	}

	on_peaceconference_started = {
		effect = {
			if = {
				limit = {
					OR = {
						AND = {
							ROOT = { has_country_flag = flag_ITA_has_italo_turkish_war_goal }
							FROM = { original_tag = TUR }
						}
						AND = {
							FROM = { has_country_flag = flag_ITA_has_italo_turkish_war_goal }
							ROOT = { original_tag = TUR }
						}
					}
				}

				log = "Peace conference between [ROOT.GetName] and [FROM.GetName] in which one participant has 'flag_ITA_has_italo_turkish_war_goal' flag and the other is Turkey. Removing the flag_ITA_has_italo_turkish_war_goal flag."

				ROOT = {
					if = {
						limit = { has_country_flag = flag_ITA_has_italo_turkish_war_goal }
						clr_country_flag = flag_ITA_has_italo_turkish_war_goal
					}
				}

				FROM = {
					if = {
						limit = { has_country_flag = flag_ITA_has_italo_turkish_war_goal }
						clr_country_flag = flag_ITA_has_italo_turkish_war_goal
					}
				}
			}
		}
	}
}
