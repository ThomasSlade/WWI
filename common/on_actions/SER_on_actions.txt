on_actions = {
	on_startup = {
		effect = {
			SER = {
				set_variable = { balkan_arbitration_opponent = BUL }
			}
			BUL = {
				set_variable = { balkan_arbitration_opponent = SER }
			}
		}
	}

	on_state_control_changed = {
		effect = {
			if = {
				limit = { FROM.FROM = { has_state_flag = flag_balkan_war_disputed } }
				effect_SER_count_balkan_disputes = yes
			}
		}
	}

	#ROOT = attacking side
	#FROM = defending side
	on_war_relation_added = {
		effect = {
			# Determine if this is the Balkan War.
			if = {
				# If the war has the Balkan League against Turkey, this is the 1st Balkan War.
				limit = {
					NOT = { check_variable = { global.first_balkan_war_state < 0 } }
					OR = {
						FROM = { original_tag = TUR }
						ROOT = { original_tag = TUR }
					}
					OR = {
						check_variable = { global.balkan_coalition_founder = FROM }
						check_variable = { global.balkan_coalition_founder = ROOT }
					}
				}

				set_variable = { global.first_balkan_war_state = 1 }

				every_country = {
					limit = { has_idea = idea_SER_balkan_coalition_member }
					country_event = { id = event_enter_first_balkan_war.1 hours = 1 }
				}

				var:global.balkan_coalition_founder = {
					news_event = { id = event_WWI_balkan_league_first_balkan_war.1 hours = 6 }
				}
			}

			# Run the 'Serbian Vojvodina Rebels against Austria' event's trigger whenever Serbia joins a war.
			if = {
				limit = {
					AND = {
						OR = {
							tag = SER
							FROM = { tag = SER }
						}
						AUS = { has_idea = AUS_vojvodina_separatists }
					}
				}
				# Attacker or defender could be Serbia/Serbia's opponent, and the event itself will filter Serbia out, so just fire to both of them.
				ROOT = { country_event = { id = event_AUS_generic.4 random_days = 2 } }
				FROM = { country_event = { id = event_AUS_generic.4 random_days = 2 } }
			}

			# Clear partnership ideas that don't make sense if the countries are fighting.
			if = {
				limit = {
					tag = SER
					FROM = { has_idea = SER_idea_danube_trade_partner }
				}
				FROM = { remove_ideas = SER_idea_danube_trade_partner }
			}
			if = {
				limit = {
					FROM = { tag = SER }
					has_idea = SER_idea_danube_trade_partner
				}
				ROOT = { remove_ideas = SER_idea_danube_trade_partner }
			}
		}
	}

	#ROOT is the winner, FROM is the loser. 
	on_peaceconference_ended = {
		effect = {
			if = {
				limit = {
					check_variable = { global.first_balkan_war_state = 1 }
					OR = {
						FROM = { original_tag = TUR }
						ROOT = { original_tag = TUR }
					}
					OR = {
						check_variable = { global.balkan_coalition_founder = FROM }
						check_variable = { global.balkan_coalition_founder = ROOT }
					}
				}
				
				log = "Ending the First Balkan War."

				set_global_flag = flag_first_balkan_war_finished

				if = {
					limit = {
						check_variable = { global.balkan_coalition_founder = ROOT }
					}

					log = "Balkan League won the First Balkan War. Initialising Balkan Disputes system."
					set_variable = { global.first_balkan_war_state = 2 }

					effect_SER_count_balkan_disputes = yes

					if = {
						limit = { has_global_flag = flag_balkan_disputes_ongoing }
						log = "As Balkan disputes were detected (flag_balkan_disputes_ongoing = true), firing the event warning Balkan League members about the system."
						every_country = {
							limit = { has_country_flag = flag_balkan_coalition_member }
							country_event = { id = event_WWI_balkan_league_first_balkan_war.2 hours = 6 }
						}
					}
					else = {
						log = "As no Balkan disputes were detected (flag_balkan_disputes_ongoing = false), not firing the event warning Balkan League members about the system."
					}
				}
				else = {
					log = "Balkan League lost First Balkan War."
					set_variable = { global.first_balkan_war_state = 3 }
				}
			}

			if = {
				# If Turkey beat a Balkan League member, remove them from the coalition (this also happens on capitulation, but I'm guessing countries don't always have to capitulate to be defeated).
				limit = {
					check_variable = { global.first_balkan_war_state = 1 }
					ROOT = { original_tag = TUR }
					FROM = { trigger_SER_is_in_balkan_coalition = yes }
				}
				FROM = {
					effect_SER_leave_balkan_coalition = yes
				}
			}

			if = {
				limit = {
					check_variable = { global.second_balkan_war_state = 0 }
					OR = {
						# Need to invert the checks since we don't know if FROM or THIS is the 2nd Balkan War instigator.
						AND = {
							var:global.second_balkan_war_attacker = { tag = FROM }
							THIS = {
								OR = {
									has_country_flag = flag_second_balkan_war_defender
									has_country_flag = flag_second_balkan_war_opportunist
								}
							}
						}
						AND = {
							var:global.second_balkan_war_attacker = { tag = THIS }
							FROM = {
								OR = {
									has_country_flag = flag_second_balkan_war_defender
									has_country_flag = flag_second_balkan_war_opportunist
								}
							}
						}
					}
				}

				log = "Peace conference ended involving Balkan War Attacker '[var:global.second_balkan_war_attacker.GetName]', and the attacker no longer had any enemies with tags 'flag_second_balkan_war_defender' or 'flag_second_balkan_war_opportunist'. Ending the 2nd Balkan War."

				if = {
					limit = { var:global.second_balkan_war_attacker = { tag = ROOT } }
					set_variable = { global.second_balkan_war_state = 4 }
				}
				else = {
					set_variable = { global.second_balkan_war_state = 3 }
				}

				news_event = { id = event_WWI_balkan_league_second_balkan_war.6 days = 2 }
			}
		}
	}

	# ROOT is capitulated. FROM is capitulator.
	on_capitulation = {
		effect = {
			if = {
				limit = { trigger_SER_is_in_balkan_coalition = yes }
				if = {
					limit = { has_idea = idea_SER_balkan_coalition_founder }
					# It may take a while, but either another country will become coalition leader or the coalition will now be disbanded. Either way, the capitulated country leaves the coalition.
					effect_SER_replace_balkan_coalition_leader = yes
				}
				else = {
					effect_SER_leave_balkan_coalition = yes
				}
			}
		}
	}

	#ROOT is new controller #FROM is old controller #FROM.FROM is state ID
	on_state_control_changed = {
		effect = {
			# Transfer state-related vojvodina statuses.
			if = {
				limit = {
					AND = {
						OR = {
							FROM.FROM = { state = 171 } # Styrmia
							FROM.FROM = { state = 232 } # Vojvodina
						}
						FROM = { has_idea = AUS_vojvodina_separatists }
					}
				}
				
				if = {
					limit = { 
						AND = {
							NOT = { FROM = { controls_state = 171 } }
							NOT = { FROM = { controls_state = 232 } }
						}
					}
					FROM = { remove_ideas = { AUS_vojvodina_separatists } }
				}

				if = {
					limit = {
						AND = {
							NOT = { original_tag = SER }
							NOT = { original_tag = SVO }
						}
					}

					add_ideas = AUS_vojvodina_separatists
				}
			}
		}
	}
}
