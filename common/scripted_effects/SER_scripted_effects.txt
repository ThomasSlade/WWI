effect_SER_form_balkan_coalition = {
	if = {
		limit = { NOT = { has_global_flag = flag_founded_balkan_coalition } }
		log = "[THIS.GetName] founded Balkan Coalition."
		set_global_flag = flag_founded_balkan_coalition
		add_ideas = { idea_SER_balkan_coalition_founder }
	}
	else = {
		log = "[THIS.GetName] cannot found Balkan Coalition because it has already been founded earlier in the game."
	}
}

effect_SER_find_new_balkan_coalition_leader = {
	log = "Attempting to find a new Balkan Coalition leader ..."

	get_sorted_scored_countries = {
		scorer = strongest_balkan_coalition_member_scorer
		array = balkan_coalition_members_by_strength
	}

	if = {
		limit = { check_variable = { balkan_coalition_members_by_strength^num > 0 } }
		log = "A strongest Balkan Coalition member was found ([?balkan_coalition_members_by_strength^0.GetName])."
		var:balkan_coalition_members_by_strength^0 = {
			country_event = { id = event_wwi_balkan_league_new_leader.1 hours = 1 }
		}
	}
	else = {
		log = "A strongest Balkan Coalition member was not found."
	}
}

effect_SER_end_balkan_coalition_if_one_or_less_members = {
	set_temp_variable = { balkan_coalition_members_count = 0 }

	every_country = {
		limit = { trigger_SER_is_in_balkan_coalition = yes }
		add_to_temp_variable = { balkan_coalition_members_count = 1 }
	}

	if = {
		limit = { check_variable = { balkan_coalition_members_count < 2 } }
		log = "Ending Balkan Coalition because there are less than 2 members left."
		effect_SER_end_balkan_coalition = yes
	}
}

effect_SER_end_balkan_coalition = {
	log = "Ending Balkan Coalition."

	set_global_flag = flag_disbanding_balkan_coalition	# This prevents recursion when removing the ideas, which would otherwise call this function again.

	# Remove all members before removing leader, as removing leader will just place the leader idea on another member.
	every_country = {
		limit = { has_idea = idea_SER_balkan_coalition_member }
		remove_ideas = { idea_SER_balkan_coalition_member }
	}

	# It should really have been removed during the process of removing members, as when membership reaches less than 2 
	every_country = {
		limit = { has_idea = idea_SER_balkan_coalition_founder }
		remove_ideas = { idea_SER_balkan_coalition_founder }
	}

	clr_global_flag = flag_disbanding_balkan_coalition
}

# Apply the effects of a 'Publish Lausanne Report' decision. PREV is Serbia, FROM is the target.
# Must be used in the format of scope [SER or Serbian Decision] = { [Target Tag] = { function = yes } }
SER_publish_lausanne_investigation_effect = {
	PREV = {
		if = {
			limit = {
				has_government = established
			}
			set_temp_variable = { temp_lausanne_drift_idea = token:SER_lausanne_established }
		}
		else_if = {
			limit = {
				has_government = expansionist
			}
			set_temp_variable = { temp_lausanne_drift_idea = token:SER_lausanne_expansionist }
		}
		else_if = {
			limit = {
				has_government = revolutionary
			}
			set_temp_variable = { temp_lausanne_drift_idea = token:SER_lausanne_revolutionary }
		}
		else = {
			log = "ERROR: Unhandled ideology of [FROM.GetName] (FROM.GetRulingIdeology) when applying SER_lausanne_ideology_drift."
		}
	}

	add_opinion_modifier = {
		target = PREV
		modifier = SER_lausanne_investigation
	}
	add_timed_idea = { idea = var:temp_lausanne_drift_idea days = 365 }
}

effect_SER_begin_balkan_disputes = {
	hidden_effect = {
		every_country = {
			limit = { trigger_SER_is_in_balkan_coalition = yes }
			add_to_array = { global.balkan_countries_with_disputes = THIS }
		}

		effect_SER_count_balkan_disputes = yes
	}
}

effect_SER_count_balkan_disputes = {
	# Clear existing variables to make debugging easier.
	clr_global_flag = flag_balkan_disputes_ongoing
	clr_global_flag = flag_balkan_disputes_resolved

	every_country = {
		limit = {
			trigger_SER_is_in_balkan_coalition = yes
		}

		set_variable = { balkan_war_disputed_states_count = 0 }
	}

	every_state = {
		limit = {
			is_on_continent = balkans
		}

		clr_state_flag = flag_balkan_war_disputed
		set_variable = { balkan_coalition_claimants = 0 }
	}

	# Begin by, for every state owned by a Balkan Coalition member, seeing if it is claimed by other members.
	# If it is, that state is disputed. It increases the dispute count of all claimants who do not own the state by 1.
	every_state = {
		limit = {
			is_on_continent = balkans
			owner = { trigger_SER_is_in_balkan_coalition = yes }
		}

		if = {
			limit = { has_state_flag = flag_balkan_war_dispute_resolved }
			set_variable = { balkan_war_disputed_states_count = 0 }
		}
		else = {
			for_each_scope_loop = {
				array = core_countries
				if = {
					limit = { trigger_SER_is_in_balkan_coalition = yes }
					PREV = { add_to_variable = { balkan_coalition_claimants = 1 } }
				}
			}

			if = {
				limit = { check_variable = { balkan_coalition_claimants > 1 } }
				log = "[THIS.GetName] is cored by [?balkan_coalition_claimants] members of the Balkan League. Marking as disputed."
				set_state_flag = flag_balkan_war_disputed
			
				for_each_scope_loop = {
					array = core_countries
					if = {
						limit = {
							trigger_SER_is_in_balkan_coalition = yes
							NOT = { owns_state = PREV }
						}
						add_to_variable = { balkan_war_disputed_states_count = 1 }
					}
				}

				set_global_flag = flag_balkan_disputes_ongoing
			}
		}
	}

	if = {
		limit = { NOT = { has_variable = global.country_with_most_balkan_war_disputes } }
		log = "No country yet considered to have the most balkan war disputes. Setting to coalition leader."
		random_country = {
			limit = { has_idea = idea_SER_balkan_coalition_founder }
			set_variable = { global.country_with_most_balkan_war_disputes = THIS }
		}
	}

	set_temp_variable = { most_disputes = global.country_with_most_balkan_war_disputes }
	var:global.country_with_most_balkan_war_disputes = {
		set_temp_variable = { most_disputes_amount = balkan_war_disputed_states_count }
	}

	# Finally, find which country has the most disputes. Mark that country as the Balkan Opponent (the one that will launch the 2nd Balkan War).
	# Unfortunately can't use a scorer for this because it doesn't support tie-breaking by using comparative country strength.
	every_country = {
		limit = { trigger_SER_is_in_balkan_coalition = yes }
	
		log = "[THIS.GetName] has [?balkan_war_disputed_states_count] disputed states from the Balkan War."
	
		if = {
			limit = {
				check_variable = {
					var = balkan_war_disputed_states_count
					value = most_disputes_amount
					compare = greater_than
				}
			}
	
			set_temp_variable = { most_disputes = THIS }
			set_temp_variable = { most_disputes_amount = THIS.balkan_war_disputed_states_count }
		}
		else_if = {
			# If some countries have equal number of disputed states, pick the more powerful to be the opponent.
			limit = {
				check_variable = {
					var = balkan_war_disputed_states_count
					value = most_disputes_amount
					compare = equals
				}
				var:most_disputes = {
					strength_ratio = {
						tag = PREV
						ratio < 1
					}
				}
			}
		
			set_temp_variable = { most_disputes = THIS }
		}
	}

	if = {
		limit = {
			check_variable = {
				var = most_disputes
				value = global.country_with_most_balkan_war_disputes 
				compare = not_equals
			}
		}
		log = "[?most_disputes.GetName] had more disputes or was stronger than [?country_with_most_balkan_war_disputes.GetName]. Making it the new Balkan League opponent."
		set_variable = { global.country_with_most_balkan_war_disputes = most_disputes }
	}

	if = {
		limit = { NOT = { has_global_flag = flag_balkan_disputes_ongoing } }
		set_global_flag = flag_balkan_disputes_resolved
	}
}

# End the Balkan Disputes system, keeping the League together and averting the 2nd Balkan War.
effect_SER_end_balkan_disputes = {
	set_global_flag = flag_SER_second_balkan_war_averted
	remove_mission = mission_SER_balkan_coalition_breakup
	effect_SER_clear_balkan_disputes = yes
	random_country = {
		limit = { has_idea = idea_SER_balkan_coalition_founder }
		country_event = { id = event_WWI_balkan_league_second_balkan_war_averted.2 hours = 1 }
	}
}

effect_SER_clear_balkan_disputes = {
	log = "([ROOT.GetName]) Clearing flags and variables of Balkan Dispute system."

	# Avoid clearing country with most balkan disputes, as that's used when picking the country that starts the 2nd Balkan War.
	clr_global_flag = flag_balkan_disputes_ongoing
	clr_global_flag = flag_balkan_disputes_resolved

	every_country = {
		limit = {
			trigger_SER_is_in_balkan_coalition = yes
		}

		clear_variable = balkan_war_disputed_states_count
	}

	every_state = {
		limit = {
			is_on_continent = balkans
		}

		clr_state_flag = flag_balkan_war_disputed
		clear_variable = balkan_coalition_claimants
	}

	clear_array = global.balkan_countries_with_disputes
}

# We track who did the most resolving so that the news event can correctly congratulate whoever saved the Balkan League :)
effect_SER_add_resolution_point = {
	add_to_variable = { balkan_resolutions_made = 1 }

	if = {
		limit = {
			OR = {
				NOT = { has_variable = global.most_balkan_resolutions_country }
				check_variable = { balkan_resolutions_made > global.most_balkan_resolutions }
			}
		}
		set_variable = { global.most_balkan_resolutions_country = THIS }
		set_variable = { global.most_balkan_resolutions = balkan_resolutions_made }
	}
}
