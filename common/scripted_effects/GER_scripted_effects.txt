effect_GER_begin_alsatian_plebiscite = {
	set_temp_variable = { param_state = ELR }
	effect_GER_remove_federal_state = yes

	if = {
		limit = { controls_state = 2 }
		set_temp_variable = { param_state = ELA }
		effect_GER_add_federal_state = yes
	}
	if = {
		limit = { controls_state = 107 }
		set_temp_variable = { param_state = LOT }
		effect_GER_add_federal_state = yes
	}

	set_global_flag = flag_alsatian_plebiscite_ongoing

	country_event = { id = event_GER_alsace_plebiscite.1 }

	FRA = {
		country_event = { id = event_GER_alsace_plebiscite.1 }
	}
}

effect_GER_end_alsatian_plebiscite = {
	set_temp_variable = { number_of_states = 0 }
	set_temp_variable = { number_of_disloyal_states = 0 }

	hidden_effect = {
		2 = {
			if = {
				limit = { is_owned_by = GER }
				add_to_temp_variable = { number_of_states = 1 }
			}
			if = {
				limit = {
					check_variable = { ELA.rr_loyalty = 1 }
				}

				add_to_temp_variable = { number_of_disloyal_states = 1 }

				set_temp_variable = { param_state = ELA }
				PREV = { effect_GER_remove_federal_state = yes }

				FRA = { set_state_owner = PREV }
				remove_core_of = GER
			}
		}

		107 = {
			if = {
				limit = { is_owned_by = GER }
				add_to_temp_variable = { number_of_states = 1 }
			}
			if = {
				limit = {
					check_variable = { LOT.rr_loyalty = 1 }
				}

				add_to_temp_variable = { number_of_disloyal_states = 1 }

				set_temp_variable = { param_state = LOT }
				PREV = { effect_GER_remove_federal_state = yes }

				FRA = { set_state_owner = PREV }
				remove_core_of = GER
			}
		}

		if = {
			limit = { check_variable = { number_of_disloyal_states = 0 } }
			set_variable = { global.alsatian_plebiscite_result = 1 }
		}
		else_if = {
			limit = { check_variable = { number_of_disloyal_states < number_of_states } }
			set_variable = { global.alsatian_plebiscite_result = 2 }
		}
		else = {
			set_variable = { global.alsatian_plebiscite_result = 3 }
		}

		if = {
			limit = { check_variable = { global.alsatian_plebiscite_result = 1 } }

			set_temp_variable = { param_state = ELA }
			effect_GER_remove_federal_state = yes
			set_temp_variable = { param_state = LOT }
			effect_GER_remove_federal_state = yes

			set_temp_variable = { param_state = ELR }
			effect_GER_add_federal_state = yes
		}

		news_event = { id = event_GER_alsace_plebiscite.2 hours = 1 }
	}

	effect_tooltip = {
		custom_effect_tooltip = GER_alsatian_plebiscite_on_complete_tt
		ELA = {
			if = {
				limit = { is_in_array = { GER.rr_countries = THIS } }
				if = {
					limit = { check_variable = { rr_loyalty = 1 } }
					custom_effect_tooltip = GER_alsatian_plebiscite_will_be_ceded_tt
				}
				else = {
					custom_effect_tooltip = GER_alsatian_plebiscite_will_not_be_ceded_tt
				}
			}
		}
		LOT = {
			if = {
				limit = { is_in_array = { GER.rr_countries = THIS } }
				if = {
					limit = { check_variable = { rr_loyalty = 1 } }
					custom_effect_tooltip = GER_alsatian_plebiscite_will_be_ceded_tt
				}
				else = {
					custom_effect_tooltip = GER_alsatian_plebiscite_will_not_be_ceded_tt
				}
			}
		}
	}

	GER = {
		remove_opinion_modifier = {
			target = FRA
			modifier = opinion_GER_bismarckian_containment
		}
	}
	FRA = {
		remove_opinion_modifier = {
			target = GER
			modifier = opinion_FRA_revanche
		}
	}

	clr_global_flag = flag_alsatian_plebiscite_ongoing
}
