effect_disliked_by_tt = {
	effect_get_opinion_tt = yes

	if = {
		limit = { check_variable = { global._tier@var:param_law < global._tier@var:param_current_law } }

		if = {
			limit = { is_in_array = { global._law_oppose_decrease@var:opposition_party = param_law } }
			effect_increase_opposition = yes
		}

		if = {
			limit = { is_in_array = { global._law_endorse_decrease@var:opposition_party = param_law } }
			effect_decrease_opposition = yes
		}
	}
	else_if = {
		limit = { check_variable = { global._tier@var:param_law > global._tier@var:param_current_law } }

		if = {
			limit = { is_in_array = { global._law_oppose_increase@var:opposition_party = param_law } }
			effect_increase_opposition = yes
		}

		if = {
			limit = { is_in_array = { global._law_endorse_increase@var:opposition_party = param_law } }
			effect_decrease_opposition = yes
		}
	}

	if = {
		limit = { is_in_array = { global._law_oppose_always@var:opposition_party = param_law } }
		effect_increase_opposition = yes
	}

	if = {
		limit = { is_in_array = { global._law_endorse_always@var:opposition_party = param_law } }
		effect_decrease_opposition = yes
	}
}

# Make the opposition more extreme that the current one for the sitting government, if possible.
# param_current_cabinet: The party we're finding the more radical opposition relative to
# param_current_opposition: The current opposition
# out_radicalised_party: The more radical opposition. Unset if none is found.
# out_switched_wings: If set, it means the radical result was not politically adjacent to the old opposition. For example, an opposition of autocrats was 'radicalised' to communists because the cabinet was conservative, and communist is further from conservative than autocratic is.
effect_radicalise_opposition = {
	log = "Attempting to pick more radical opposition for [THIS.GetName], current cabinet [?param_current_cabinet.GetTokenKey] and current opposition [?param_current_opposition.GetTokenKey]."

	set_temp_variable = { param_party = param_current_cabinet }
	effect_get_party_index = yes
	set_temp_variable = { cabinet_index = out_index }

	set_temp_variable = { param_party = param_current_opposition }
	effect_get_party_index = yes
	set_temp_variable = { opposition_index = out_index }

	# If positive, the opposition is left of the cabinet. If negative, it's right of the cabinet.
	set_temp_variable = { overton_distance = opposition_index }
	subtract_from_temp_variable = { overton_distance = cabinet_index }

	set_temp_variable = { max_party_index = global._all_parties^num }
	subtract_from_temp_variable = { max_party_index = 1 }

	set_temp_variable = { next_party_index = opposition_index }

	if = {
		limit = { check_variable = { overton_distance < 0 } }
		add_to_temp_variable = { next_party_index = -1 }
	}
	else = {
		add_to_temp_variable = { next_party_index = 1 }
	}

	if = {
		limit = {
			OR = {
				check_variable = { next_party_index < 0 }
				check_variable = { next_party_index > max_party_index }
			}
		}
		log = "No"

		set_temp_variable = { next_party_index = overton_distance }
		multiply_temp_variable = { next_party_index = -1 }
		add_to_temp_variable = { next_party_index = cabinet_index }
		if = {
			limit = { check_variable = { overton_distance < 0 } }
			add_to_temp_variable = { next_party_index = 1 }
		}
		else = {
			add_to_temp_variable = { next_party_index = -1 }
		}

		log = "Ran out of room to increment opposition so trying to reverse. Overton distance was [?overton_distance]. Negated and added to cabinet index is [?next_party_index]"

		if = {
			limit = {
				OR = {
					check_variable = { next_party_index < 0 }
					check_variable = { next_party_index > max_party_index }
				}
			}

			log = "Could not find a party more radical than [?param_current_opposition.GetTokenKey] is to [?param_current_cabinet]"
		}
		else = {
			set_temp_variable = { out_radicalised_party = global._all_parties^next_party_index }
			set_temp_variable = { out_switched_wings = 1 }

			log = "Found radicalised party [?out_radicalised_party.GetTokenKey] (reversed extremism direction relative to cabinet)"
		}
	}
	else = {
		set_temp_variable = { out_radicalised_party = global._all_parties^next_party_index }

		log = "Found radicalised party [?out_radicalised_party.GetTokenKey] (incremented from previous opposition)"
	}
}

effect_select_leader = {
	if = {
		limit = { trigger_government_uses_monarch = yes }

		if = {
			limit = { NOT = { check_variable = { country_leader = var:monarch } } }
			var:monarch = {
				custom_effect_tooltip = newline_tt
				custom_effect_tooltip = newline_tt
				effect_promote_character_to_all_ideologies = yes
			}
		}
	}
	else = {
		if = {
			limit = { NOT = { check_variable = { country_leader = var:leader@var:param_party } } }
			var:leader@var:param_party = {
				custom_effect_tooltip = newline_tt
				custom_effect_tooltip = newline_tt
				effect_promote_character_to_all_ideologies = yes
			}
		}
	}
}

effect_select_leader_using_current_government_type = {
	set_temp_variable = { param_government_type = law_government }
	effect_select_leader = yes
}

effect_select_leader_using_current_cabinet = {
	set_temp_variable = { param_party = cabinet_party }
	effect_select_leader = yes
}
