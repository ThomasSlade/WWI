		}
	}

# param_law: The law we're changing to.
# Needs to be recursive so that tooltips still work even though we want to iterate over all the parties (for-loops don't support tooltips).
effect_get_opposition_effects_oppose_increase = {
	set_temp_variable = { param_party = global._all_parties^iter }

	if = {
		limit = { is_in_array = { global._law_oppose_increase@var:param_party = param_law } }

		meta_effect = {
			PARTY_NAME = "[?param_party.GetTokenKey]"

			text = {
				custom_effect_tooltip = {
					localization_key = law_oppose_increase_outline_tt
					OPPOSING_PARTY = "[PARTY_NAME]_colored"
				}
			}
		}
	}

	add_to_temp_variable = { iter = 1 }
	if = {
		limit = { check_variable = { iter < global._all_parties^num } }
		effect_get_opposition_effects_oppose_increase = yes
	}
}

# param_law: The new law
# param_current_law: The current law
effect_disliked_by_tt = {
	effect_get_opinion_tt = yes

	if = {
		limit = { check_variable = { global._tier@var:param_law < global._tier@var:param_current_law } }

		if = {
			limit = { is_in_array = { global._law_oppose_decrease@var:opposition_party = param_law } }
			effect_increase_opposition = yes
		}

		if = {
			limit = { is_in_array = { global._law_endorse_decrease@var:opposition_party = param_law } }
			effect_decrease_opposition = yes
		}
	}
	else_if = {
		limit = { check_variable = { global._tier@var:param_law > global._tier@var:param_current_law } }

		if = {
			limit = { is_in_array = { global._law_oppose_increase@var:opposition_party = param_law } }
			effect_increase_opposition = yes
		}

		if = {
			limit = { is_in_array = { global._law_endorse_increase@var:opposition_party = param_law } }
			effect_decrease_opposition = yes
		}
	}

	if = {
		limit = { is_in_array = { global._law_oppose_always@var:opposition_party = param_law } }
		effect_increase_opposition = yes
	}

	if = {
		limit = { is_in_array = { global._law_endorse_always@var:opposition_party = param_law } }
		effect_decrease_opposition = yes
	}
}

effect_increase_opposition = {
	if = {
		limit = { check_variable = { opposition_level < 4 } }
		custom_effect_tooltip = increase_opposition_tt
		
		hidden_effect = {
			set_temp_variable = { next_opposition_level = opposition_level }
			add_to_temp_variable = { next_opposition_level = 1 }

			meta_effect = {
				text = {
					add_ideas = opp_[PARTY_TYPE]_[NEXT_OPPOSITION_LEVEL]
				}
				PARTY_TYPE = "[?opposition_party.GetTokenKey]"
				NEXT_OPPOSITION_LEVEL = "[?next_opposition_level]"
			}
		}
	}
}

effect_decrease_opposition = {
	if = {
		limit = { check_variable = { opposition_level > 0 } }
		custom_effect_tooltip = decrease_opposition_tt

		hidden_effect = {
			set_temp_variable = { next_opposition_level = opposition_level }
			add_to_temp_variable = { next_opposition_level = -1 }

			meta_effect = {
				text = {
					add_ideas = opp_[PARTY_TYPE]_[NEXT_OPPOSITION_LEVEL]
				}
				PARTY_TYPE = "[?opposition_party.GetTokenKey]"
				NEXT_OPPOSITION_LEVEL = "[?next_opposition_level]"
			}
		}
	}
