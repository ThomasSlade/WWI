# Expects temp variable of name 'param_law' and value of that law's token
# Tiers determine what is considered an 'increase' or 'decrease' in a law, allowing a party to oppose the law moving in a given direction.
# e.g. the Autocrats won't mind 'increasing' from Disarmed Nation to Volunteer Only, but they will mind 'decreasing' from Limited Conscription to Volunteer Only.
effect_get_law_tier = {
	set_temp_variable = { out_law_tier = -1 }

	################################################################### Manpower
	if = {
		limit = { check_variable = { param_law = token:disarmed_nation } }
		set_temp_variable = { out_law_tier = 0 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:volunteer_only } }
		set_temp_variable = { out_law_tier = 1 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:national_service } }
		set_temp_variable = { out_law_tier = 2 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:limited_conscription } }
		set_temp_variable = { out_law_tier = 3 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:extensive_conscription } }
		set_temp_variable = { out_law_tier = 4 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:deployed_workers_0 } }
		set_temp_variable = { out_law_tier = 5 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:deployed_workers_1 } }
		set_temp_variable = { out_law_tier = 6 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:deployed_labourers_0 } }
		set_temp_variable = { out_law_tier = 5 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:deployed_labourers_1 } }
		set_temp_variable = { out_law_tier = 6 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:scraping_the_barrel } }
		set_temp_variable = { out_law_tier = 7 }
	} ################################################################### Economic
	else_if = {
		limit = { check_variable = { param_law = token:civilian_economy } }
		set_temp_variable = { out_law_tier = 0 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:low_economic_mobilisation } }
		set_temp_variable = { out_law_tier = 1 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:partial_economic_mobilisation } }
		set_temp_variable = { out_law_tier = 2 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:profit_seizure_0 } }
		set_temp_variable = { out_law_tier = 3 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:profit_seizure_1 } }
		set_temp_variable = { out_law_tier = 4 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:nationalisation_0 } }
		set_temp_variable = { out_law_tier = 3 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:nationalisation_1 } }
		set_temp_variable = { out_law_tier = 4 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:private_collaboration_0 } }
		set_temp_variable = { out_law_tier = 3 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:private_collaboration_1 } }
		set_temp_variable = { out_law_tier = 4 }
	} ################################################################### Rationing
	else_if = {
		limit = { check_variable = { param_law = token:no_rationing } }
		set_temp_variable = { out_law_tier = 0 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:price_controls } }
		set_temp_variable = { out_law_tier = 1 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:government_confiscation_0 } }
		set_temp_variable = { out_law_tier = 2 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:government_confiscation_1 } }
		set_temp_variable = { out_law_tier = 3 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:government_distribution_0 } }
		set_temp_variable = { out_law_tier = 2 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:government_distribution_1 } }

	else_if = {

	else_if = {
		limit = { check_variable = { param_law = token:wage_controls_1 } }

	}

		set_temp_variable = { out_law_tier = 4 }
	}
	else = {
		log = "ERROR: No law tier specified for law [?param_law.GetTokenKey]."
	}
}

# Figures out what the law tiers are at the game start (the on_add behaviour won't be run at the start of a game so we have to do it manually here)
effect_determine_law_tiers = {
	for_each_loop = {
		array = global.manpower_laws
		value = l
		break = break

		if = {
			limit = {
				meta_trigger = {
					text = {
						has_idea = [LAW_TOKEN]
					}

					LAW_TOKEN = "[?l.GetTokenKey]"
				}
			}
			set_temp_variable = { param_law = l }
			effect_get_law_tier = yes
			set_temp_variable = { break = 1 }
		}
	}
	set_variable = { law_tier_manpower = out_law_tier }

	for_each_loop = {
		array = global.economic_laws
		value = l
		break = break

		if = {
			limit = {
				meta_trigger = {
					text = {
						has_idea = [LAW_TOKEN]
					}

					LAW_TOKEN = "[?l.GetTokenKey]"
				}
			}
			set_temp_variable = { param_law = l }
			effect_get_law_tier = yes
			set_temp_variable = { break = 1 }
		}
	}
	set_variable = { law_tier_economic = out_law_tier }

	for_each_loop = {
		array = global.rationing_laws
		value = l
		break = break

		if = {
			limit = {
				meta_trigger = {
					text = {
						has_idea = [LAW_TOKEN]
					}

					LAW_TOKEN = "[?l.GetTokenKey]"
				}
			}
			set_temp_variable = { param_law = l }
			effect_get_law_tier = yes
			set_temp_variable = { break = 1 }
		}
	}
	set_variable = { law_tier_rationing = out_law_tier }
	

	for_each_loop = {
		array = global.workers_rights_laws
		value = l
		break = break

		if = {
			limit = {
				meta_trigger = {
					text = {
						has_idea = [LAW_TOKEN]
					}

					LAW_TOKEN = "[?l.GetTokenKey]"
				}
			}
			set_temp_variable = { param_law = l }
			effect_get_law_tier = yes
			set_temp_variable = { break = 1 }
		}
	}
	set_variable = { law_tier_workers_rights = out_law_tier }

	for_each_loop = {
		array = global.civil_rights_laws
		value = l
		break = break

		if = {
			limit = {
				meta_trigger = {
					text = {
						has_idea = [LAW_TOKEN]
					}

					LAW_TOKEN = "[?l.GetTokenKey]"
				}
			}
			set_temp_variable = { param_law = l }
			effect_get_law_tier = yes
			set_temp_variable = { break = 1 }
		}
	}
	set_variable = { law_tier_civil_rights = out_law_tier }

	for_each_loop = {
		array = global.blockade_policy_laws
		value = l
		break = break

		if = {
			limit = {
				meta_trigger = {
					text = {
						has_idea = [LAW_TOKEN]
					}

					LAW_TOKEN = "[?l.GetTokenKey]"
				}
			}
			set_temp_variable = { param_law = l }
			effect_get_law_tier = yes
			set_temp_variable = { break = 1 }
		}
	}
	set_variable = { law_tier_blockade_policy = out_law_tier }
}

effect_law_change_opposition = {

			effect_get_law_tier = yes

			effect_disliked_by_tt = yes

			if = {
				limit = { check_variable = { opposition_party = out_opposing_party } }

# Given a party and opposition set see if a law is opposed, e.g. for 'aurocratic' party and if 'decreasing' the law, is this law opposed?
# This also creates tooltips outlining who opposes what.
# param_party: token of the party you're checking opposes this law
# param_opinion_set: token of the opposition set (i.e. for law decreases, law increases, etc.) to check
# param_law: the law
effect_is_law_in_opinion_set = {
	log = "Checking if law [?param_law.GetTokenKey] is in opinion set [?param_opinion_set.GetTokenKey]_[?param_party.GetTokenKey]"
	set_temp_variable = { out_is_opposed_by = 0 }

	meta_effect = {
		OPPOSITION_SET = "[?param_opinion_set.GetTokenKey]"
		PARTY = "[?param_party.GetTokenKey]"

		text = {
			if = {
				limit = { is_in_array = { array = [OPPOSITION_SET]_[PARTY] value = param_law } }
				custom_effect_tooltip = {
					localization_key = [OPPOSITION_SET]_outline_tt
					OPPOSING_PARTY = "[PARTY]_colored"
				}
				set_temp_variable = { out_is_opposed_by = 1 }
				log = "IT IS!"
			}
		}
	}
}

# Needs to be recursive so that tooltips still work even though we want to iterate over all the parties (for-loops don't support tooltips).
effect_get_opposition_effects_recursive = {
	set_temp_variable = { param_party = all_parties^iter }

	effect_is_law_in_opinion_set = yes
	if = {
		limit = {
			check_variable = { out_is_opposed_by = 1 }
			check_variable = { opposition_party = param_party }
		}

		set_temp_variable = { out_law_in_opinion_set = 1 }
	}

	add_to_temp_variable = { iter = 1 }
	if = {
		limit = { check_variable = { iter < all_parties^num } }
		effect_get_opposition_effects_recursive = yes
	}
}

effect_get_opposition_effects = {
	set_temp_variable = { iter = 0 }
	effect_get_opposition_effects_recursive = yes
}

# param_tier: the tier of the law being adopted
# param_current_tier: the tier of the current law being replaced
effect_disliked_by_tt = {
	# If the law in question would be a 'decrease' of the current law:
	if = {
		limit = { check_variable = { param_tier < param_current_tier } }

		set_temp_variable = { param_opinion_set = token:law_oppose_decrease }
		effect_get_opposition_effects = yes
		if = {
			limit = { check_variable = { out_law_in_opinion_set > 0 } }
			effect_increase_opposition = yes
		}
		set_temp_variable = { out_law_in_opinion_set = 0 }

		set_temp_variable = { param_opinion_set = token:law_endorse_decrease }
		effect_get_opposition_effects = yes
		if = {
			limit = { check_variable = { out_law_in_opinion_set > 0 } }
			effect_decrease_opposition = yes
		}
		set_temp_variable = { out_law_in_opinion_set = 0 }
	}
	# If the law in question would be an 'increase' of the current law:
	else_if = {
		limit = { check_variable = { param_tier > param_current_tier } }

		set_temp_variable = { param_opinion_set = token:law_oppose_increase }
		effect_get_opposition_effects = yes
		if = {
			limit = { check_variable = { out_law_in_opinion_set > 0 } }
			effect_increase_opposition = yes
		}
		set_temp_variable = { out_law_in_opinion_set = 0 }
		log = "Just set out_law_in_opinion_set to 0 and its actual value is [?out_law_in_opinion_set]"

		set_temp_variable = { param_opinion_set = token:law_endorse_increase }
		effect_get_opposition_effects = yes
		if = {
			limit = { check_variable = { out_law_in_opinion_set > 0 } }
			log = "Decreasing opposition because out_law_in_opinion_set = [?out_law_in_opinion_set]"
			effect_decrease_opposition = yes
		}
		set_temp_variable = { out_law_in_opinion_set = 0 }
	}

	# Some law effects happen whether a law is increasing or decreasing in tier (or staying the same)
	set_temp_variable = { param_opinion_set = token:law_oppose_always }
	effect_get_opposition_effects = yes
	if = {
		limit = { check_variable = { out_law_in_opinion_set > 0 } }
		effect_increase_opposition = yes
	}
	set_temp_variable = { out_law_in_opinion_set = 0 }

	set_temp_variable = { param_opinion_set = token:law_endorse_always }
	effect_get_opposition_effects = yes
	if = {
		limit = { check_variable = { out_law_in_opinion_set > 0 } }
		effect_decrease_opposition = yes
	}
	set_temp_variable = { out_law_in_opinion_set = 0 }
}

effect_increase_opposition = {
	custom_effect_tooltip = increase_opposition_tt
	
	hidden_effect = {
		set_temp_variable = { next_opposition_level = opposition_level }
		add_to_temp_variable = { next_opposition_level = 1 }

		meta_effect = {
			text = {
				add_ideas = opp_[PARTY_TYPE]_[NEXT_OPPOSITION_LEVEL]
			}
			PARTY_TYPE = "[?opposition_party.GetTokenKey]"
			NEXT_OPPOSITION_LEVEL = "[?next_opposition_level]"
		}
	}
}

	hidden_effect = {

		meta_effect = {

			}
			PARTY_TYPE = "[?opposition_party.GetTokenKey]"
			NEXT_OPPOSITION_LEVEL = "[?next_opposition_level]"
		}
	}
