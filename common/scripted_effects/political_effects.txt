# Expects temp variable of name 'param_law' and value of that law's token
# Tiers determine what is considered an 'increase' or 'decrease' in a law, allowing a party to oppose the law moving in a given direction.
# e.g. the Autocrats won't mind 'increasing' from Disarmed Nation to Volunteer Only, but they will mind 'decreasing' from Limited Conscription to Volunteer Only.
effect_get_law_tier = {
	set_temp_variable = { out_law_tier = -1 }

	################################################################### Manpower
	if = {
		limit = { check_variable = { param_law = token:disarmed_nation } }
		set_temp_variable = { out_law_tier = 0 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:volunteer_only } }
		set_temp_variable = { out_law_tier = 1 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:limited_conscription } }
		set_temp_variable = { out_law_tier = 2 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:extensive_conscription } }
		set_temp_variable = { out_law_tier = 3 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:deployed_workers_0 } }
		set_temp_variable = { out_law_tier = 4 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:deployed_workers_1 } }
		set_temp_variable = { out_law_tier = 5 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:deployed_labourers_0 } }
		set_temp_variable = { out_law_tier = 4 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:deployed_labourers_1 } }
		set_temp_variable = { out_law_tier = 5 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:scraping_the_barrel } }
		set_temp_variable = { out_law_tier = 6 }
	} ################################################################### Economic
	else_if = {
		limit = { check_variable = { param_law = token:civilian_economy } }
		set_temp_variable = { out_law_tier = 0 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:low_economic_mobilisation } }
		set_temp_variable = { out_law_tier = 1 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:partial_economic_mobilisation } }
		set_temp_variable = { out_law_tier = 2 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:profit_seizure_0 } }
		set_temp_variable = { out_law_tier = 3 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:profit_seizure_1 } }
		set_temp_variable = { out_law_tier = 4 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:nationalisation_0 } }
		set_temp_variable = { out_law_tier = 3 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:nationalisation_1 } }
		set_temp_variable = { out_law_tier = 4 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:private_collaboration_0 } }
		set_temp_variable = { out_law_tier = 3 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:private_collaboration_1 } }
		set_temp_variable = { out_law_tier = 4 }
	} ################################################################### Rationing
	else_if = {
		limit = { check_variable = { param_law = token:no_rationing } }
		set_temp_variable = { out_law_tier = 0 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:price_controls } }
		set_temp_variable = { out_law_tier = 1 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:government_confiscation_0 } }
		set_temp_variable = { out_law_tier = 2 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:government_confiscation_1 } }
		set_temp_variable = { out_law_tier = 3 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:government_distribution_0 } }
		set_temp_variable = { out_law_tier = 2 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:government_distribution_1 } }
		set_temp_variable = { out_law_tier = 3 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:harsh_rationing } }
		set_temp_variable = { out_law_tier = 4 }
	} ################################################################### Workers' Rights
	else_if = {
		limit = { check_variable = { param_law = token:forced_labour } }
		set_temp_variable = { out_law_tier = 0 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:quitting_illegal } }

	}
	else_if = {
		limit = { check_variable = { param_law = token:worker_collaboration_1 } }
		set_temp_variable = { out_law_tier = 4 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:wage_controls_0 } }
		set_temp_variable = { out_law_tier = 3 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:wage_controls_1 } }
		set_temp_variable = { out_law_tier = 4 }
	} ################################################################### Civil Rights
	else_if = {
		limit = { check_variable = { param_law = token:entrenched_nationalism } }
		set_temp_variable = { out_law_tier = 0 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:ethnic_rights_0 } }
		set_temp_variable = { out_law_tier = 1 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:ethnic_rights_1 } }
		set_temp_variable = { out_law_tier = 2 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:ethnic_rights_2 } }
		set_temp_variable = { out_law_tier = 3 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:domestic_rights_0 } }
		set_temp_variable = { out_law_tier = 1 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:domestic_rights_1 } }
		set_temp_variable = { out_law_tier = 2 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:domestic_rights_2 } }
		set_temp_variable = { out_law_tier = 3 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:women_minority_rights } }
		set_temp_variable = { out_law_tier = 4 }
	} ################################################################### Blockade Policy
	else_if = {

	}
	else_if = {
		limit = { check_variable = { param_law = token:surface_blockade_1 } }
		set_temp_variable = { out_law_tier = 3 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:submarine_blockade_0 } }
		set_temp_variable = { out_law_tier = 2 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:submarine_blockade_1 } }
		set_temp_variable = { out_law_tier = 3 }
	}
	else_if = {
		limit = { check_variable = { param_law = token:unrestricted_blockade } }
		set_temp_variable = { out_law_tier = 4 }
	}
	else = {
		log = "ERROR: No law tier specified for law [?param_law.GetTokenKey]."
	}
}

# Figures out what the law tiers are at the game start (the on_add behaviour won't be run at the start of a game so we have to do it manually here)
effect_determine_law_tiers = {
	for_each_loop = {
		array = global.manpower_laws
		value = l
		break = break

		if = {
			limit = {
				meta_trigger = {
					text = {
						has_idea = [LAW_TOKEN]
					}

					LAW_TOKEN = "[?l.GetTokenKey]"
				}
			}
			set_temp_variable = { param_law = l }
			effect_get_law_tier = yes
			set_temp_variable = { break = 1 }
		}
	}
	set_variable = { law_tier_manpower = out_law_tier }
	log = "[THIS.GetName] had law [?l.GetTokenKey] of tier [?out_law_tier]"

	for_each_loop = {
		array = global.economic_laws
		value = l
		break = break

		if = {
			limit = {
				meta_trigger = {
					text = {

		}
	}
	set_variable = { law_tier_economic = out_law_tier }
	log = "[THIS.GetName] had law [?l.GetTokenKey] of tier [?out_law_tier]"

	for_each_loop = {

		array = global.workers_rights_laws
		value = l
		break = break

		if = {
			limit = {
				meta_trigger = {
					text = {
						has_idea = [LAW_TOKEN]
					}

					LAW_TOKEN = "[?l.GetTokenKey]"
				}
			}
			set_temp_variable = { param_law = l }
			effect_get_law_tier = yes
			set_temp_variable = { break = 1 }
		}
	}
	set_variable = { law_tier_workers_rights = out_law_tier }
	log = "[THIS.GetName] had law [?l.GetTokenKey] of tier [?out_law_tier]"

		array = global.civil_rights_laws
		value = l

# Param: param_law must be set to the token of the law calling this.
# This method has a lot of boilerplate because meta_effects don't seem to work well with tooltips.
effect_disliked_by_tt = {
	# 1 = opposes decrease, 2 = opposes increase, 3 = dislikes in general
	set_temp_variable = { will_increase_opposition = 0 }

	log = "Creating Opposition tooltip for law [?param_law.GetTokenKey], tier [?param_tier], current tier [?param_current_tier]"

	# If the law in question would be a 'decrease' of the current law:
	if = {
		limit = { check_variable = { param_tier < param_current_tier } }
		log = "Querying for a decrease in law tier"
		if = {
			limit = { is_in_array = { array = law_oppose_decrease_autocratic value = param_law } }
			custom_effect_tooltip = {
				localization_key = decrease_opposed_by_tt
				OPPOSING_PARTY = "autocratic_colored"
			}
			if = {
				limit = { check_variable = { opposition_party = token:autocratic } }
				set_temp_variable = { will_increase_opposition = 1 }
			}
		}
		if = {
			limit = { is_in_array = { array = law_oppose_decrease_conservative value = param_law } }
			custom_effect_tooltip = {
				localization_key = decrease_opposed_by_tt
				OPPOSING_PARTY = "conservative_colored"
			}
			if = {
				limit = { check_variable = { opposition_party = token:conservative } }
				set_temp_variable = { will_increase_opposition = 1 }
			}
		}
		if = {
			limit = { is_in_array = { array = law_oppose_decrease_liberal value = param_law } }
			custom_effect_tooltip = {
				localization_key = decrease_opposed_by_tt
				OPPOSING_PARTY = "liberal_colored"
			}
			if = {
				limit = { check_variable = { opposition_party = token:liberal } }
				set_temp_variable = { will_increase_opposition = 1 }
			}
		}
		if = {

				localization_key = decrease_opposed_by_tt

				limit = { check_variable = { opposition_party = token:socialist } }
				set_temp_variable = { will_increase_opposition = 1 }
			}

				localization_key = decrease_opposed_by_tt

	else_if = {

			limit = { is_in_array = { array = law_oppose_increase_autocratic value = param_law } }
			custom_effect_tooltip = {
				localization_key = increase_opposed_by_tt
				OPPOSING_PARTY = "autocratic_colored"
			}
			if = {
				limit = { check_variable = { opposition_party = token:autocratic } }
				set_temp_variable = { will_increase_opposition = 2 }
			}
		}
		if = {
			limit = { is_in_array = { array = law_oppose_increase_conservative value = param_law } }
			custom_effect_tooltip = {
				localization_key = increase_opposed_by_tt
				OPPOSING_PARTY = "conservative_colored"
			}
			if = {
				limit = { check_variable = { opposition_party = token:conservative } }
				set_temp_variable = { will_increase_opposition = 2 }
			}
		}
		if = {
			limit = { is_in_array = { array = law_oppose_increase_liberal value = param_law } }
			custom_effect_tooltip = {
				localization_key = increase_opposed_by_tt

			}
		}
		if = {
			limit = { is_in_array = { array = law_oppose_increase_communist value = param_law } }
			custom_effect_tooltip = {
				localization_key = increase_opposed_by_tt
				OPPOSING_PARTY = "communist_colored"
			}
			if = {
				limit = { check_variable = { opposition_party = token:communist } }
				set_temp_variable = { will_increase_opposition = 2 }
			}
		}
	}

	# These dislikes are ALWAYS applied no matter if we're increasing or decreasing the law.
	if = {
		limit = { is_in_array = { array = disliked_laws_autocratic value = param_law } }
		custom_effect_tooltip = {
			localization_key = always_opposed_by_tt
			OPPOSING_PARTY = "autocratic_colored"

			limit = { check_variable = { opposition_party = token:autocratic } }

		}
		if = {
			limit = { check_variable = { opposition_party = token:socialist } }
			set_temp_variable = { will_increase_opposition = 3 }

	}
	if = {
		limit = { is_in_array = { array = disliked_laws_communist value = param_law } }
		custom_effect_tooltip = {
			localization_key = always_opposed_by_tt
			OPPOSING_PARTY = "communist_colored"
		}
		if = {
			limit = { check_variable = { opposition_party = token:communist } }
			set_temp_variable = { will_increase_opposition = 3 }
		}
	}

	if = {
		limit = { check_variable = { will_increase_opposition > 0 } }
		set_temp_variable = { param_opposition_type = will_increase_opposition }
		effect_increase_opposition = yes
	}
