# param_alignment: The tokenized ideology this country will be assigned the leader of.
effect_set_great_power_alignment = {
	set_country_flag = flag_great_power

	custom_effect_tooltip = effect_become_great_power_tt
	custom_effect_tooltip = effect_set_great_power_alignment_tt
	set_variable = { leader_of_alignment = param_alignment }
	add_popularity = {
		ideology = var:param_alignment
		popularity = 0.2
	}

	hidden_effect = {
		news_event = { id = event_wwi_become_great_power.3 days = 1 }
	}
}

# param_booster: The tag of the country doing the boosting.
effect_PREV_boost_ideology = {
	meta_effect = {
		ALIGNMENT_BOOST_MODIFIER = "modifier_boost_party_[?param_booster:current_party_ideology_group.GetTokenKey]"
		text = {
			add_dynamic_modifier = {
				modifier = [ALIGNMENT_BOOST_MODIFIER]
				scope = var:param_booster
			}
		}
	}

	var:param_booster = {
		add_dynamic_modifier = {
			modifier = modifier_boosting_party
			scope = PREV
		}

		add_to_array = { boosting_alignment_of = PREV }
	}
}

# param_booster: The tag of the country doing the boosting.
effect_PREV_cancel_boost_ideology = {
	meta_effect = {
		ALIGNMENT_BOOST_MODIFIER = "modifier_boost_party_[?param_booster:current_party_ideology_group.GetTokenKey]"
		text = {
			remove_dynamic_modifier = {
				modifier = [ALIGNMENT_BOOST_MODIFIER]
				scope = var:param_booster
			}
		}
	}

	var:param_booster = {
		remove_dynamic_modifier = {
			modifier = modifier_boosting_party
			scope = PREV
		}

		if = {
			limit = { NOT = { is_in_array = { boosting_alignment_of = PREV } } }
			log = "WARNING: cancel_boost_ideology was called by [THIS.GetName] to stop boosting [PREV.GetName]. However, [PREV.GetName] was not in the boosting_alignment_of array of [THIS.GetName]. This should not be possible."
		}
		remove_from_array = { boosting_alignment_of = PREV }
	}
}
