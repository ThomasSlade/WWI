AUS_sway_free_city_to_austria = {
	custom_effect_tooltip = AUS_sway_free_city
	if = {
		limit = { has_idea = AUS_italian_influenced_free_city }
		log = "Swaying [GetName] to Austria: Had Italian influence"
		remove_ideas = AUS_italian_influenced_free_city
		# Needed for tooltip purposes. -1 for Italy, 1 for Austria.
		set_variable = { previous_influence = -1 }
	}
	else_if = {
		limit = {
			NOT = {
				OR = {
					has_idea = AUS_austrian_influenced_free_city
					has_idea = AUS_italian_influenced_free_city
				}
			}
		}
		log = "Swaying [GetName] to Austria: Had no influence"
		add_ideas = AUS_austrian_influenced_free_city
		set_variable = { previous_influence = 0 }
	}
}

# Reports how the free city's influence level has just been changed, for tooltip purposes.
AUS_sway_free_city_tt = {
	if = {
		limit = { has_idea = AUS_austrian_influenced_free_city }
		add_ideas = AUS_austrian_influenced_free_city
	}
	else_if = {
		limit = { has_idea = AUS_italian_influenced_free_city }
		add_ideas = AUS_italian_influenced_free_city
	}
	else_if = {
		limit = { check_variable = { previous_influence = 1 } }
		remove_ideas = AUS_austrian_influenced_free_city
	}
	else_if = {
		limit = { check_variable = { previous_influence = -1 } }
		remove_ideas = AUS_italian_influenced_free_city
	}
}

				value = ZAR
			}
		}
	}
}

# Check state local organisation bonuses for all employed characters.
AUS_update_ethnicity_local_organisation = {
	every_character = {
		limit = { is_in_active_role = yes }
		AUS_add_ethnicity_local_organisations = yes
	}
}

# Root should be the character.
AUS_add_ethnicity_local_organisations = {
	effect_tooltip = {
		if = {
			limit = { owner = { has_completed_focus = AUS_request_national_representatives } }
			custom_effect_tooltip = AUS_request_national_representatives_unlocked_tt
		}
		else = {
			custom_effect_tooltip = AUS_request_national_representatives_locked_tt
		}
		
		if = {
			limit = { has_character_flag = ethnicity_czechoslovak }
			custom_effect_tooltip = AUS_czechoslovak_local_organisation_tt
		}
		if = {
			limit = { has_character_flag = ethnicity_croatslovene }
			custom_effect_tooltip = AUS_croatslovene_local_organisation_tt
		}
		if = {
			limit = { has_character_flag = ethnicity_galician }
			custom_effect_tooltip = AUS_polish_local_organisation_tt
		}
		if = {
			limit = { has_character_flag = ethnicity_ruthene }
			custom_effect_tooltip = AUS_ruthene_local_organisation_tt
		}
		add_dynamic_modifier = { modifier = AUS_ethnic_local_organisation }
	}

	hidden_effect = {
		if = {
			limit = {
				owner = { has_completed_focus = AUS_request_national_representatives }
			}
			if = {
				limit = { has_character_flag = ethnicity_czechoslovak }
				log = "[THIS.GetName] was a valid character for providing czechoslovak AUS_ethnic_local_organisation bonuses."
				every_state = {
					limit = { is_core_of = CZE is_controlled_by = owner }
					if = {
						add_dynamic_modifier = { modifier = AUS_ethnic_local_organisation }
					}
				}
			}
			if = {
				limit = { has_character_flag = ethnicity_croatslvoene }
				log = "[THIS.GetName] was a valid character for providing south slav AUS_ethnic_local_organisation bonuses."
				every_state = {
					limit = { is_core_of = SCS is_controlled_by = owner 
					}
					if = {
						add_dynamic_modifier = { modifier = AUS_ethnic_local_organisation }
					}
				}
			}
			if = {
				limit = { has_character_flag = ethnicity_galician }
				log = "[THIS.GetName] was a valid character for providing polish AUS_polish_local_organisation bonuses."
				every_state = {
					limit = { is_core_of = POL is_controlled_by = owner  }
					if = {
						add_dynamic_modifier = { modifier = AUS_ethnic_local_organisation }
					}
				}
			}
			if = {
				limit = { has_character_flag = ethnicity_ruthene }
				log = "[THIS.GetName] was a valid character for providing ukrainian AUS_ethnic_local_organisation bonuses."
				every_state = {
					limit = { is_core_of = UKR is_controlled_by = owner  }
					if = {
						add_dynamic_modifier = { modifier = AUS_ethnic_local_organisation }
					}
				}
			}
		}
	}
}

effect_AUS_foment_obrenovic_sympathies = {
	add_to_variable = { obrenovic_sympathies_level = 1 }
	add_to_variable = { times_fomented_sympathies = 1 }
	# Start counting the order of fomenting from 1, not 0, because 0 doesn't set off the has_flag trigger.
	FROM = {
		log = "Fomenting Obrenovic sympathies in [?THIS.GetName]. Order of fomentation is [?PREV.times_fomented_sympathies]."
		set_state_flag = obrenovic_sympathies
		set_variable = { fomented_sympathies_order = PREV.times_fomented_sympathies }
	}
	

	set_temp_variable = { level_plus_one = obrenovic_sympathies_level }
	add_to_temp_variable = { level_plus_one = 1 }

	set_temp_variable = { required_infantry_eq = 500 }
	multiply_temp_variable = { required_infantry_eq = level_plus_one }

	set_temp_variable = { required_support_eq = 50 }
	multiply_temp_variable = { required_support_eq = level_plus_one }

	add_equipment_to_stockpile = {
		type = infantry_equipment
		amount = required_infantry_eq
	}

	add_equipment_to_stockpile = {
		type = support_equipment
		amount = required_support_eq
	}

	FROM = { custom_effect_tooltip = state_will_side_with_obrenovic_tt }

	hidden_effect = {
		# Clear the countdown to removing a sympathetic state.
		remove_mission = mission_AUS_obrenovic_sympathies_waning

		FROM = {
			save_event_target_as = obrenovic_sympathiser_province
			controller = {
				country_event = { id = events_AUS_foment_obrenovic_sympathies.2 hours = 1 }
			}
		}
	}
}

	}
	else_if = {
		limit = { state = 135 }

			limit = { PREV = { state = 224 } }
			build_railway = {
				level = 3
				start_province = 5254
				path = { 6157 5877 5254	5230 5217 5229 5316 5488 5518 4639 4564 4907 }
			}
		}
		else_if = {
			limit = { PREV = { state = 164 } }
			# Mitrovica to Kotor via Cetinje
			log = "Montenegro to Mitrovica"
			build_railway = {
				level = 3
				path = { 6157 5877 5254	5184 5698 5699 5455 5746 5422 4768 4615 }
			}

			log = "ERROR: [PREV.GetName] is not a valid destination state for building the Novi Pazar line."
		}
	}
	else = {
		log = "ERROR: [THIS.GetName] is not a valid route state for building the Novi Pazar line."
	}
}

# Determines strength of the Obrenovic coup given how many core states have Obrenovic sympathies, and stores as variable on the ROOT scope named 'obrenovic_coup_strength'
effect_AUS_determine_obrenovic_coup_strength = {
	set_temp_variable = { obrenovic_states = 0 }
	for_each_scope_loop = {
		array = owned_controlled_states
		log = "[THIS.GetName]"
		if = {
			limit = { has_state_flag = obrenovic_sympathies }
			add_to_temp_variable = { obrenovic_states = 1 }
		}
	}

	set_temp_variable = { obrenovic_coup_strength = obrenovic_states }
	divide_variable = { obrenovic_coup_strength = num_owned_controlled_states }
	log = "Determined Obrenovic coup strength as [?obrenovic_states] Obrenovic States / Core States [?num_owned_controlled_states] = [?obrenovic_coup_strength]
}
