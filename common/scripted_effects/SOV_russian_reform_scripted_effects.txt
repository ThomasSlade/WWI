### Tsar Favor Resource Effects ###
effect_SOV_spend_1_tsar_favor = {
	custom_effect_tooltip = tsar_favor_spend_1_tt
	subtract_from_variable = { tsar_favor = 1 }
	if = {
		limit = { check_variable = { tsar_favor < 0 } }
		log = "ERROR: After spending tsar_favor, the remaining tsar_favor was negative. Do not call this effect when there isn't enough tsar favour to spend."
	}
}

# The conditional effects only apply the effect if Russia is has autocratic tsardom idea.
effect_SOV_spend_1_tsar_favor_if_tsarist = {
	if = {
		limit = {
			has_idea = idea_SOV_autocratic_tsardom
		}
		effect_SOV_spend_1_tsar_favor = yes
	}
}
