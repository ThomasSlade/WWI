# For a given disputed state, gets how many countries would be offered concessions to resolve the dispute (the owner doesn't get a concession, and the resolver never makes a concession to themselves).
# FROM is the disputed state, ROOT is the country doing the resolving.
trigger_SER_get_balkan_dispute_concession_recipients = {
	set_temp_variable = { recipients = FROM.balkan_coalition_claimants }
	add_to_temp_variable = { recipients = -1 }

	if = {
		limit = {
			FROM = {
				is_core_of = ROOT
				NOT = { is_controlled_by = ROOT }
			}
		}
		add_to_temp_variable = { recipients = -1 }
	}

	always = yes
}

trigger_SER_is_serb = {
	OR = {
		tag = SER
		tag = MNT
		tag = VOJ
		# Not counting SCS because it's only partly Serbian
	}
}

trigger_SER_is_in_balkan_coalition = {
	OR = {
		has_idea = idea_SER_balkan_coalition_founder
		has_idea = idea_SER_balkan_coalition_member
	}
}

trigger_SER_can_join_first_balkan_war = {
 	trigger_SER_is_in_balkan_coalition = yes
	NOT = { has_country_flag = flag_joioned_first_balkan_war }
	check_variable = { global.first_balkan_war_state = 1 }
}
