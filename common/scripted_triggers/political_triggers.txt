	}

# param_law: The potential new law
trigger_opposition_allows_law_change = {
	meta_trigger = {
		LAW_TYPE = "[?global._law_type@var:param_law.GetTokenKey]"

		text = {
			set_temp_variable = { param_current_law = law_[LAW_TYPE] }
		}
	}

	custom_trigger_tooltip = {
		tooltip = will_not_exceed_opposition_limit_tt
		if = {
			limit = { check_variable = { opposition_level > 2 } }

			NOT = {
				AND = {
					check_variable = { global._tier@var:param_law < global._tier@var:param_current_law }
					is_in_array = { global._law_oppose_decrease@var:opposition_party = param_law }
				}
				AND = {
					check_variable = { global._tier@var:param_law > global._tier@var:param_current_law }
					is_in_array = { global._law_oppose_increase@var:opposition_party = param_law }
				}
				is_in_array = { global._law_oppose_always@var:opposition_party = param_law }
			}
		}
	}
}

	OR = {
		check_variable = { param_government_type = token:gov_absolute_monarchy }
		check_variable = { param_government_type = token:gov_constitutional_monarchy }
	}
}

	set_temp_variable = { param_government_type = government_type }
	OR = {
		has_idea = gov_absolute_monarchy
		has_idea = gov_constitutional_monarchy
	}
}

# param_party: The party we're checking if they can be in cabinet.
trigger_government_can_have_party_in_cabinet = {
	custom_trigger_tooltip = {
		tooltip = government_can_have_party_in_cabinet_tt
		if = {
			limit = {
				OR = {
					has_idea = gov_military_dictatorship
					has_idea = gov_absolute_monarchy
					has_idea = gov_constitutional_monarchy
				}
			}
			NOT = { check_variable = { param_party = token:communist } }
		}
		if = {
			limit = {
				OR = {
					has_idea = gov_military_dictatorship
					has_idea = gov_absolute_monarchy
				}
			}
			NOT = { check_variable = { param_party = token:socialist } }
		}
	}
}
