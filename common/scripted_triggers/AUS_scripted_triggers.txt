trigger_AUS_can_foment_obrenovic_sympathies = {
	custom_trigger_tooltip = {
		tooltip = trigger_AUS_at_least_one_serbian_state_not_obrenovic

		set_temp_variable = { non_obrenovic_states = 0 }
		SER = {
			all_core_state = {
				if = {
					limit = {
						is_controlled_by = SER
						NOT = { has_state_flag = obrenovic_sympathies }
					}
					add_to_temp_variable = { non_obrenovic_states = 1 }
				}
			}
		}
		check_variable = { non_obrenovic_states > 1 }
	}
	

	FROM = {
		NOT = { has_state_flag = obrenovic_sympathies }
		any_neighbor_state = {
			OR = {
				controls_or_subject_of = yes
				has_state_flag = obrenovic_sympathies
			}
		}
	}

	set_temp_variable = { level_plus_one = obrenovic_sympathies_level }
	add_to_temp_variable = { level_plus_one = 1 }

	set_temp_variable = { required_infantry_eq = 500 }
	multiply_temp_variable = { required_infantry_eq = level_plus_one }
	add_to_temp_variable = { required_infantry_eq = -1 }

	set_temp_variable = { required_support_eq = 50 }
	multiply_temp_variable = { required_support_eq = level_plus_one }
	add_to_temp_variable = { required_support_eq = -1 }

	has_equipment = { infantry_equipment > required_infantry_eq }
	has_equipment = { support_equipment > required_support_eq }
}
